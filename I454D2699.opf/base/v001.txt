བསྟོད་སྨད་ཆགས་སྡང་ལ་སོགས་པའི་རྐྱེན་མང་བར་བརྟེན་ཆོས་ཡང་དག་པར་སྒྲུབ་པ་ལ་བར་ཆད་མང་པོ་ཡོང་གི་ཡོད། ང་རང་ཚོ་ཁྱིམ་ན་གནས། འཁོར་བའི་ལོངས་སྤྱོད་སྒྲུབ། ཚེ་འདིའི་བདེ་སྐྱིད་དང༌། ཚེ་འདིའི་དོན་དུ་བྱ་བ་བྱེད་དགོས་དུས་ཆགས་སྡང་ཁོང་ཁྲོ་ང་རྒྱལ་སོགས་ལ་གཞི་མ་བཅོལ་ཐབས་མེད། དཔེར་ན་ཁྱིམ་ཚང་ནང་དུ་སྡོད་པའི་ཚེ་ནང་མིའི་རྙོག་དྲ་དང༌། ཕྱི་ལ་འགྲོ་བའི་ཚེ་རྐྱེན་ངན་མང་པོ་ཞིག་འཕྲད་ཡོང་དུས། ང་ཚོས་བླ་མའི་མདུན་ནས་ཆོས་ཞུས། ཆོས་ཉམས་ལེན་བྱེད་ཀྱི་ཡོད་པའི་རྣམ་པ་ཞིག་བྱེད་བཞིན་ཡོད། ཡིན་ན་ཡང་རྐྱེན་ངན་ཞིག་དང་འཕྲད་དུས་ཆོས་ནས་གསུང་བ་ཇི་བཞིན་དུ་ཡང་དག་པར་རྒྱུད་ཐོག་དུ་ཉམས་སུ་ལེན་གྱི་མེད། མ་ཎི་འདོན་གྱིན་འདོན་གྱིན་ཁོང་ཁྲོ་སྐྱེ། བཛྲ་གུ་རུ་བགྲང་གིན་བགྲང་གིན་ཁོང་ཁྲོ་སྐྱེ། ཁ་ནས་སྐྱབས་འགྲོ་འདོན་གྱིན་འདོན་གྱིན་ཁོང་ཁྲོ་དང་ཞེ་སྡང་དང་ཕྲག་དོག་སྐྱེ་བ། ལུས་ངག་གིས་དགེ་བ་སྒྲུབ་བཞིན་དུ་སེམས་ཀྱི་ནང་ཆགས་སྡང་ཁོང་ཁྲོ་ང་རྒྱལ་སོགས་དང་མཉམ་དུ་སྐྱེ་བཞིན་ཡོད། ལུས་ཀྱིས་ཕྱག་འཚལ། སྐོར་བ་ཕྱིན། ལག་གཡས་པས་མ་ཎི་འཁོར་ལོ་བསྐོར། གཡོན་མས་ཕྲེང་བ་བགྲང་བ་སོགས་རྣམ་འགྱུར་གྱི་ཐོག་ནས་དགེ་བ་སྒྲུབ་པའི་ཚུལ་དུ་སྣང་ཡང་དེ་དང་དུས་མཉམ་དུ་སེམས་ནང་ཁོང་ཁྲོ་དང་འདོད་ཆགས། ང་རྒྱལ་དང་ཕྲག་དོག་སོགས་རང་རྒྱལ་བ་དང་གཞན་ཕམ་པ་སོགས་བསམ་མནོ་མང་པོ་འཁོར་ཡོང་མཁན་དེ་འདྲའི་ཆོས་པ་དེ། ཆོས་རྣམ་པར་དག་པ་ལམ་དུ་མ་ཕྱིན་པའི་མཚན་མ་རེད། དེ་བཞིན་དུ་ངག་ནས་སྐྱབས་འགྲོ་གསོལ་འདེབས་སོགས་འདོན་པ་ཡིན་ན་ཡང༌། དེ་དང་དུས་མཉམ་དུ་སེམས་ཀྱི་ནང་ལས་སུ་མ་རུང་བའི་ཉོན་མོངས་པའི་རྣམ་པར་རྟོག་པ་མང་པོ་ཞིག་འཁོར་ཡོང་ཆོག་གི་ཡོད་དུས། དེའི་སྟོབས་ཀྱིས་ངག་གི་དགེ་བ་དེ་དགེ་བ་རྣམ་དག་ལ་འགྲོ་མི་ཐུབ། འོ་ན་ལུས་ཀྱི་དགེ་བ་དང་ངག་གི་དགེ་བ་དེ་དགེ་བ་རྣམ་པར་དག་པ་ལ་འགྲོ་དང་མ་འགྲོ་བའི་བྱེད་ལས་གཙོ་བོ་དེ་གང་གི་བྱེད་ཅེ་ན། གཙོ་བོ་སེམས་ཡིན། སེམས་ཀྱི་དགེ་བ་དེ་སྟོབས་ཆེ་བའོ། །གང་ཞེ་ན། དེ་ཡང་དད་པ། མོས་གུས། མི་རྟག་པ་བསྒོམ་པའི་སེམས། བདག་མེད་བསྒོམ་པའི་སེམས། བྱང་ཆུབ་ཀྱི་སེམས། སྙིང་རྗེའི་སེམས། བྱམས་པའི་སེམས་ལ་སོགས་པ་ཡིད་ཀྱི་དགེ་བ་དེ་དག་ཉམས་སུ་ལེན་པའི་ཚེ། དེ་དང་དུས་མཉམ་དུ་སེམས་ཀྱི་ནང་མི་དགེ་བའི་ལས་སུ་མ་རུང་བའི་རྣམ་རྟོག་སྐྱེ་བའི་གོ་སྐབས་གཏན་ནས་མེད་པའི་ཕྱིར་ན། ཡིད་ཀྱི་དགེ་བ་དེ་སྟོབས་ཆེ་བ་རེད། ལུས་དང་ངག་གི་དགེ་བ་དེ་སྟོབས་ཆུང་ཆུང་རེད། གལ་ཏེ་ཡིད་ཀྱི་དགེ་བ་དང་ལུས་ངག་གི་དགེ་བ་ལྷན་དུ་འགྲོ་ན། ལུས་ངག་ཡིད་གསུམ་གའི་དགེ་བ་དེ་རྣམ་པར་དག་པ་ལ་འགྲོ་གི་ཡོད། དེས་ན་ང་ཚོས་སེམས་ཀྱི་ནང་བག་ཡོད་དྲན་ཤེས་བསྟེན་པའི་ཐོག་ནས་དགེ་བ་རྣམ་པར་དག་པ་སྒྲུབ་དགོས། ལྷག་པར་དུ་ལས་དང་པོ་པར་རྣམ་རྟོག་གི་སྟོབས་ཀྱིས་རང་ལ་རང་དབང་མེད་པར་གྱུར་པས་ན། དབེན་པ་བསྟེན་དགོས་པར་གསུངས། ལུས་དབེན་པ་དང་སེམས་དབེན་པ་ཞེས་གཙོ་བོ་གཉིས་ཡོད། དང་པོ་ལུས་དབེན་པ་ལའང་བྲག་ཕུག་དང་གཡའ་རི། གངས་རི་དང་རྫ་རི། ནགས་ཁྲོད་དང་རྫ་ཁྲོད་ལ་སོགས་པ་ལ་བསྡད་ན། གནས་དབེན་པ་ཡིན་གསུངས། གནས་དབེན་པ་ལ་བསྟེན་པ་ཡིན་ན། རང་ལ་བར་ཆད་བྱེད་མཁན་གྱི་རྐྱེན་ངན་པ་ལས་ཐག་རིང་དུ་སོང༌། དགེ་བ་སྒྲུབ་པ་ལ་མཐུན་རྐྱེན་དང་དགེ་བ་སྒྲུབ་པ་ལ་དཀའ་ངལ་མེད་པ་ཡོང་གི་ཡོད་དུས་གནས་དབེན་པ་བསྟེན་དགོས། ང་ཚོས་མཚམས་བསྡད་ཟེར་བའི་སྐབས་སུ་སྐབས་འགར་ལྷ་དྲག་པོ་ཞིག་སྒྲུབ་དགོས་ན། ནག་ཁུང་ཞིག་གི་ནང་ལ་འཛུལ། སྒོ་སྒེ་ཁུང་ཕར་བརྒྱབ། སུས་ཀྱང་མཐོང་ས་མེད་པ་ལྟ་བུ་ཞིག་ལ་མཚམས་ཞེས་ངོས་འཛིན་བྱེད་ཀྱི་ཡོད། དེ་མ་རེད། མཚམས་བསྡད་དགོས་པའི་རྒྱུ་མཚན་དེ་ལུས་ཀྱི་འགུལ་བསྐྱོད་དང་ངག་གི་སྨྲ་བརྗོད་སོགས་དགེ་སྦྱོར་ལ་བར་ཆད་བྱེད་ཀྱི་ཡོད་དུས། ལུས་དབེན་པ། ངག་དབེན་པ། སེམས་དབེན་པའི་སྒོ་ནས་དབེན་པར་བསྡད། ཉོན་མོངས་པ་ལས་ཐག་སྲིང༌། ཉོན་མོངས་པ་ལས་ཐག་རིང་དུ་བསྡད། ཉོན་མོངས་སྐྱེ་བའི་རྐྱེན་ཞིག་ལ་བརྟེན་ནས་ཁོང་ཁྲོ་འདོད་ཆགས་སོགས་སྐྱེ་ཡི་ཡོད། དཔེར་ན། གཟུགས་མཛེས་པོ་ཞིག་མཐོང་དུས་སེམས་ལ་ཆགས། གཟུགས་མི་མཛེས་པ་ཞིག་མཐོང་དུས་མ་དགའ་བར་འགྱུར། །སྒྲ་སྙན་པོ་ཞིག་ཐོས་དུས་སེམས་ལ་ཆགས། སྒྲ་མ་སྙན་པ་ཞིག་ཐོས་དུས་སེམས་འཁྲུག་པར་འགྱུར་བ་སོགས་ཆགས་སྡང་སྐྱེ་བའི་རྐྱེན་གཙོ་བོ་དེ་ཡུལ་གྱི་རྐྱེན་ལ་བརྟེན་ནས་ཡོང་གི་ཡོད། ལས་དང་པོ་པས་ཐད་ཀར། ཉོན་མོངས་རང་བཞིན་གྱིས་མ་གྲུབ་པ་རེད། རྣམ་རྟོག་རང་བཞིན་གྱིས་མ་གྲུབ་པ་རེད། འདོད་ཆགས་ཞེ་སྡང་གཏི་མུག་རང་བཞིན་གྱིས་གྲུབ་ཡོད་མ་རེད་ཅེས་བསམ་དགོས་ན། བསམ་མི་ཐུབ། བསམ་པ་ཡིན་ན་ཡང་དེའི་སྟོབས་ཀྱིས་ཉོན་མོངས་མགོ་གནོན་ཐུབ་པར་ཁག་པོ་རེད། དེར་བརྟེན་དྭགས་པོ་རིན་པོ་ཆེས། ལས་དང་པོ་པས་ཉོན་མོངས་དང་འཐབ་པའི་དུས་མིན། འབྲོས་པའི་དུས་ཡིན་གསུངས། ཉོན་མོངས་པ་སྐྱེ་བའི་རྐྱེན་ལས་ཐག་རིང་བའི་གངས་ཁྲོད། རི་ཁྲོད། རྫ་ཁྲོད། ནགས་ཁྲོད་སོགས་ལ་རང་གཅིག་པུར་བསྡད་ན། ཁོང་ཁྲོ་སྐྱེ་བའི་རྐྱེན་མེད། ཕྲག་དོག་སྐྱེ་བའི་རྐྱེན་མེད། ཆགས་པ་ལ་སོགས་པ་ལས་སུ་མ་རུང་བའི་རྣམ་རྟོག་སྐྱེ་བའི་རྐྱེན་ལས་ཐག་རིང་དུ་སོང་བས་ཇི་ཙམ་གནས་དབེན་པ་ལ་གནས་པ་དེ་ཙམ་གྱིས་དགེ་སྦྱོར་འཕེལ་བ་ལ་ཕན་ཐོགས་ཆེན་པོ་ཡོད། འདི་ལས་དང་པོ་པའི་ཉམས་ལེན་གྱི་སྐབས་ལ་རེད། རྟོགས་པ་ཚད་མཐོན་པོ་ལ་སླེབས། ཉམས་ལེན་ཚད་མཐོན་པོ་ལ་སླེབས་པ་ཡིན་ན་ནི། རི་ཁྲོད་ལ་བསྡད་ནའང་འདྲ། ཚོགས་པའི་ནང་ལ་བསྡད་ནའང་འདྲ། གྲོང་ཁྱེར་ལ་བསྡད་ན་ཡང་འདྲ་ཁྱད་པར་ཡོད་མེད་དེ་སེམས་ལ་དབང་ཐོབ་ཟིན་པ་དང༌། སྒྲོ་འདོགས་སེམས་ལ་ཆོད་ཟིན་པའི་ཕྱིར། སེམས་ལ་དབང་ཐོབ་པ་ཡིན་ན་རྣལ་འབྱོར་པ་ཟེར་བ་རེད། དེ་ལ་ཕྱིའི་རྐྱེན་གྱིས་བར་ཆད་ཀྱིས་གནོད་མི་ཐུབ། ལས་དང་པོ་པའི་ཉམས་ལེན་པ་རི་ཁྲོད་སོགས་བསྡད་མ་ཐུབ་ན་ཡང༌། མཚམས་རེར་སོ་སོའི་ཁྱིམ་ནང་ཡིན་ན་ཡང་མཚམས་བསྡད། སྒོ་སྒེ་ཁུང་སོགས་ཕར་རྒྱབ། རང་གཅིག་པུར་ཉིན་གསུམ་བཞིའམ་བདུན་ཕྲག་རེ་ཙམ་བསྡད་པ་ཡིན་ནའང་ཕན་ཐོགས་ཆེན་པོ་ཡོད། དེ་ལ་དགོས་པ་ཆེན་པོ་ཡོད། དེས་ན་གནས་དབེན་པ་བསྟེན་དགོས་ཞེས་གསུངས། རྒྱལ་སྲས་ལག་ལེན་ནང༌།
ཡུལ་ངན་སྤངས་པས་ཉོན་མོངས་རིམ་གྱིས་འགྲིབ། །
རྣམ་གཡེང་མེད་པས་དགེ་སྦྱོར་ངང་གིས་འཕེལ། །
རིག་པ་དྭངས་པས་ཆོས་ལ་ངེས་ཤེས་སྐྱེ། །
དབེན་པ་བསྟེན་པ་རྒྱལ་སྲས་ལག་ལེན་ཡིན། །
ཞེས་གསུངས་པ་ལྟར། ཡུལ་ངན་སྤངས་ཞེས་པའི་ཡུལ་ངན་པ་དེ་མགོ་ལ་རྭ་ཅོ་སྐྱེས་པ། མཆེ་བ་གཙིགས་པ། བཟོ་ལྟ་སྡུག་པོ། དྲེགས་པ་ལྡན་པ་ལ་ཟེར་གྱི་མེད། ཡུལ་གང་ཞིག་ལ་བརྟེན་ནས་དེའི་རྐྱེན་གྱིས་རང་གི་སེམས་དབང་མེད་པར་བཟོ་མཁན་ཞིག་ཡོད་ན། དེ་ལ་ཡུལ་ངན་པ་དང་རྐྱེན་ངན་པ་བརྗོད་པ་ཡིན། ད་ལྟ་རང་སེམས་སྐྱིད་པོ་ཡོད་པ། བློ་ཆོས་སུ་གང་ལ་གང་ཙམ་འགྲོ་ཐུབ། སྐྱབས་གནས་དཀོན་མཆོག་གསུམ་ལ་དད་པ་ཡང་གང་ལ་གང་ཙམ་ཡོད་པ། རིན་ཆེན་བྱང་ཆུབ་ཀྱི་སེམས་ཀྱང་གང་ལ་གང་ཙམ་བློ་སྦྱོང་ཐུབ་མཁན་ཞིག་ཡོད་པ་ཡིན་ན་ཡང༌། རྐྱེན་ངན་པ་ཞིག་དང་འཕྲད་ཡོང་དུས་སྔར་གྱི་དད་པ་དང་བྱམས་སྙིང་རྗེ་ལ་སོགས་ཡོད་པ་དེ་ཡང་མར་ཉམས་པར་བྱེད་མཁན། སེམས་ལས་སུ་མ་རུང་བར་བྱེད་མཁན། སྐྱིད་པོ་མེད་པར་བཟོ་མཁན། སེམས་ལྷོད་པོ་མེད་པར་བཟོ་མཁན། སེམས་ལ་རང་དབང་མེད་པར་བཟོ་མཁན་ཞིག་ཡོད་ན། དེ་ལ་ཡུལ་ངན་པ་ཟེར་བ་རེད། ཡུལ་ངན་དེ་སྤངས་ཡོང་དུས་ཉོན་མོངས་སྐྱེ་བའི་རྐྱེན་ལས་ཐག་རིང་དུ་སོང་བས་ཉོན་མོངས་དེ་དག་རིམ་གྱིས་མར་འགྲིབ། དང་པོ་ཉོན་མོངས་སྟོབས་ཆེན་ཞིག་ཡོད། གང་ཡིན་ཞེ་ན། དེ་རྟག་ཏུ་ང་རང་ཚོའི་རྐྱེན་ལ་གནས་བསྡད་ཡོད་པ་རེད། ཞོགས་པ་སྔ་དྲོ་ཡར་ལྡང་ནས་བཟུང་སྟེ་ཕྱི་དྲོ་ས་མ་རུབ་ཀྱི་བར་ཉིན་གང་རྐྱེན་ངན་པ་ཞིག་ལ་བརྟེན་ནས་འགྲོ་ཡི་ཡོད་དུས་ནམ་ཡང་སེམས་ལས་སུ་རུང་བར་གྱུར་ཐུབ་ཀྱི་མེད། དགེ་བ་སྒྲུབ་པའི་ཚུལ་བྱེད་ཀྱི་ཡོད་ཀྱང༌། དགེ་བ་སྒྲུབ་ཀྱིན་སྒྲུབ་ཀྱིན་ལ་དེ་དང་དུས་མཉམ་དུ་སེམས་ལས་སུ་མ་རུང་བ་མང་པོ་ཞིག་ཡོང་གི་ཡོད། དེའི་ཕྱིར་ན་ཡུལ་ངན་སྤང་དགོས་རེད། ཡུལ་ངན་པ་སྤངས་ན་དེའི་རྐྱེན་གྱིས་སེམས་རང་དབང་མེད་པར་བཟོ་བའི་གོ་སྐབས་མེད་པ་ཡིན་དུས། ཉོན་མོངས་དེ་དག་རིམ་གྱིས་རིམ་གྱིས་མཐོན་པོ་ཡོད་པ་དེ་དམའ་རུ་འགྲོ། མང་པོ་ཡོད་པ་དེ་ཉུང་དུ་འགྲོ། རིང་པོ་ཡོད་པ་དེ་མར་ཐུང་དུ་འགྲོ་ཐུབ་པའི་དཔེ་བཞིན་དུ་མར་ཉུང་དུ་ཉུང་དུ་འགྲིབ་ཀྱི་རེད་ཅེས་གསུངས། རྣམ་གཡེང་མེད་པས་དགེ་སྦྱོར་ངང་གིས་འཕེལ་ཞེས་རྐྱེན་ཡུལ་ངན་པ་དང་བྲལ་ཡོང་དུས་རྣམ་པར་གཡེང་བའི་གཞི་མེད་པ་རེད། ང་ཚོ་དུས་རྒྱུན་འདོད་ཆགས་ཀྱི་རྣམ་པར་རྟོག་པའི་རྗེས་སུ་སོང་ནས་སེམས་རྣམ་པར་གཡེང༌། ཞེ་སྡང་གི་རྣམ་པར་རྟོག་པའི་རྗེས་སུ་སོང་ནས་སེམས་རྣམ་པར་གཡེང༌། ཕྲག་དོག་དང་ང་རྒྱལ་དང་སེར་སྣ་ལ་སོགས་པ་གཙོ་བོར་གྱུར་པའི་མི་དགེ་བ་ཕྱིན་ཅི་ལོག་པའི་རྣམ་པར་རྟོག་པ་གྲངས་མང་བ་ཚད་མེད་པ་ཞིག་ཡོད། དེ་དག་གི་རྗེས་སུ་སེམས་རྟག་ཏུ་རྣམ་པར་གཡེང་དུས་གསུང་རབ་ལ་ཐོས་པ་བྱེད་རྒྱུའི་མོས་པ་མེད། བསམ་པར་བྱེད་པ་ལ་ཡང་མོས་པ་མེད། སྒོམ་པ་ལ་ཡང་མོས་པ་མེད། བྱམས་སྙིང་རྗེ་བྱང་ཆུབ་ཀྱི་སེམས་དང་བདག་མེད་པ་དང་མི་རྟག་པ་ལ་སོགས་པ་སྒོམ་པ་ལ་ཡང་མོས་པ་མེད། ང་ཚོས་དགེ་བ་སྒྲུབ་པའི་སྐབས་སུ་དགེ་བ་དེ་མ་སྒྲུབ་རང་སྒྲུབ་རེད། མ་བསྒྲུབས་ན་འགྲིག་གི་མི་འདུག་སྙམ་ཏེ་ལྐུགས་པ་ཨུ་ཚུགས་ཀྱི་ཐོག་ནས་དགེ་བ་སྒྲུབ་པ་ཞིག་འགྱུརད་ཀྱི་འདུག དཔེར་ན་དཔེ་ཆ་ཞིག་བལྟ། སློབ་སྦྱོང་ཞིག་བྱེད་པའི་སྐབས་སུ་ཆུ་ཚོད་ཕྱེད་ཀའམ་ཆུ་ཚོད་གཅིག་ཙམ་རིང་ལ་ཡང་རྒྱབ་ན་བ། མདུན་ན་བ། འོག་ནས་ན་བ། མགོ་བོ་ན་བ། རྔུལ་ནག་རྒྱག་པ། གཉིད་ཁུག་པ་སོགས་ཀྱི་དཀའ་ངལ་པོ་ཞིག་ཡོང་གི་ཡོད། འོན་ཀྱང་གཟུགས་མཐོང་བརྙན་འཕྲིན་ལྟ་བ་ལ་ནི་ཆུ་ཚོད་ནས་ཆུ་ཚོད་བསྡད་ཐུབ་པ། ཐོ་རངས་ཆུ་ཚོད་གསུམ་པ་བཞི་པའི་བར་བལྟ་ཐུབ་པ། འདི་དུས་ཕོ་བ་ཡང་ལྟོགས་རྒྱུ་མེད་པ། ཁ་ཡང་སྐོམ་རྒྱུ་མེད་པ། ཆུང་དང་ཇ་སོགས་ཀྱང་འཐུང་མི་དགོས་པ་མ་ཟད། གཉིད་ཀྱང་བྲོ་རྒྱུ་མེད་པ་ཡོད། དེ་གང་རེད་ཅེ་ན། ལས་སུ་མ་རུང་བའི་སེམས་ཞེས་ཐོག་མེད་ནས་བག་ཆགས་ངན་པ་ལ་གོམས་པའི་སྟོབས་ཀྱི་འབྲས་རྟགས་རེད། རང་རྒྱུད་ལ་ཉོན་མོངས་ཟད་མ་ཐུབ་པ་དང་ཉོན་མོངས་སྟོབས་ཆེན་པོ་ཡོད་པའི་སྟོབས་ཀྱིས་ང་ཚོ་དགེ་བ་སྒྲུབ་པ་ལ་སེམས་ཤུགས་ཆུང་བ། མི་དགེ་བའི་ལས་གསོག་པ་ལ་སེམས་ཤུགས་ཆེ་བ། ཆགས་སྡང་གི་ལས་གསོག་པ་ལ་དཀའ་ངལ་མེད་པ། དེ་དང་བྲལ་བའི་མ་ཆགས་ཞེ་སྡང་གཏི་མུག་མེད་པའི་དགེ་བ་སྒྲུབ་པ་ལ་དཀའ་ངལ་ཆེ་བ། ཆགས་པ་དང་ཞེ་སྡང་གཏི་མུག་གི་ལས་ལ་སྤྲོ་བ་ཡོད་ལ་དཀའ་ངལ་མེད་པ་ཅན་ཞིག་རང་བཞིན་གྱིས་ཡོང་གི་ཡོད། དཔེར་ན། ཁོང་ཁྲོ་ཞིག་སྐྱེ་ཡོང་དུས་ཁོང་ཁྲོའི་སྟོབས་ཀྱིས་རང་རྒྱུད་ལས་སུ་མ་རུང་བར་བཟོས། ཁྲོང་ཁྲོའི་སྟོབས་ཀྱིས་རང་རྒྱུད་བཀྲ་མ་ཤིས་པར་བཟོས། དེའི་འབྲས་བུར་ཕྱི་མ་དམྱལ་བ་སོགས་ལ་འགྲོ་ཡི་ཡོད་པ་ཤེས་ཀྱི་ཡོད། ཞེ་སྡང་ཡག་པོ་མིན་པ་ང་ཚོས་ཤེས་ཀྱི་ཡོད། ཕྲག་དོག་དང་ང་རྒྱལ་གྲས་ཡག་པོ་མིན་པ་སུས་ཀྱང་ཤེས་མོད། ཕལ་ཆེར་ཆོས་ལ་སློབ་སྦྱོང་བྱས་མ་མྱོང་ན་ཡང་ནང་པའི་གྲལ་དུ་ཚུད་ཙམ་ཞིག་གིས་ཀྱང་བསམ་མནོ་འདི་གཏོང་ཤེས་ཀྱི་རེད། ནང་པ་མིན་པ་ཞིག་གིས་ཀྱང་ཞེ་སྡང་དང་ཁོང་ཁྲོ་སོགས་ཡག་པོ་མིན་པ་དང་སྐྱིད་པོ་མེད་པར་བཟོ་ཡི་ཡོད་པ་ཤེས་ཀྱི་ཡོད། ཆོས་ཤེས་མཁན་ཞིག་གིས་ནི་ཞེ་སྡང་དང་ཁོང་ཁྲོ་ཡག་པོ་མེད་པ། ཚེ་འདི་ལ་དགྲ་མང་པོ་འབྱུང་བ་དང་ཚེ་ཕྱི་མ་ལ་དམྱལ་སོགས་ལ་འགྲོ་བ་ཤེས་ཀྱི་ཡོད་རེད་དེ། ཡིན་ན་ཡང་ཐོག་མེད་ནས་དེ་ལ་གོམས་པ་དང་བག་ཆགས་ཤུགས་ཆེ་བའི་དབང་གིས་ཞེ་སྡང་དང་ཁོང་ཁྲོ་སོགས་སྐྱོན་ཡིན་པར་ཤེས་བཞིན་དུ་ད་དུང་ཡང་དེ་ལ་འཛེམས་པ་མེད་པ་ངལ་བ་མེད་པའི་སྒོ་ནས་ངམ་ངམ་ཤུགས་ཤུགས་ཀྱིས་འཇུག་གི་ཀྱི་ཡོད། དེའི་ཕྱིར་ན་སེམས་རྣམ་གཡེང་མེད་པར་བྱ་རྒྱུ་དེ་གལ་ཆེ་ཤོས་རེད། ང་རང་ཚོ་ལས་དང་པོ་པ་ཡིན་ན་ཡང༌། ལས་དང་པོ་པ་ཞེས་ལབ་པ་རེད་མ་གཏོགས་གཟབ་གཟབ་བྱེད་ཀྱི་ཡིན་ན། རྟག་པར་ལས་དང་པོ་པ་ལ་གནས་ཀྱི་ཡོད་མ་རེད། ལས་དང་པོ་པ་ཡིན་པའི་དབང་དུ་བཏང་བ་ཡིན་ན་ཡང་གང་ལྟར་ཆོས་ཉམས་ལེན་པ་ཚོས་རྒྱུན་དུ་ཆགས་སོགས་ཉོན་མོངས་པའི་རྣམ་པར་རྟོག་པ་སྐྱེ་ཡོང་དུས་ཉིད་ནས་ཉོན་མོངས་ཟེར་བ་དེ་འདི་རེད། ཞེ་སྡང་ཡོང་གི་འདུག །ཕྲག་དོག་དང༌། ང་རྒྱལ་གྱི་རྣམ་རྟོག་སྐྱེ་ཡོང་གི་འདུག་སྙམ་སྟེ་རྣམ་རྟོག་ སྐྱེས་མ་ཐག་ཏུ་གཉེན་པོའི་སྟོབས་ཀྱིས་ངོས་ཟིན་པར་བྱེད་རྒྱུ་ཤིན་ཏུ་གལ་ཆེ། དང་པོ་ངོས་ཟིན་དགོས་པ་རེད། ཉོན་མོངས་སྐྱེས་ཡོང་དུས་སྐྱེ་བཅུག །ཕྲག་དོག་དང་ཞེ་སྡང་ལ་སོགས་པ་སྐྱེས་ཡོང་དུས་དེའི་དབང་གིས་རང་ཉིད་ཕར་དབང་མེད་དུ་འགྲོ། འདོད་ཆགས་དང་ཞེ་སྡང་དང་ཕྲག་དོག་སོགས་ཀྱི་རྣམ་པར་རྟོག་པས་རང་ཉིད་རང་དབང་མེད་པར་ངེས་མེད་དུ་འདྲེན་ཏེ་རང་ཉིད་གཡོག་པོ་ལྟ་བུ་དང་ཉོན་མོངས་པ་དེ་རྒྱལ་པོ་ལྟ་བུར་གྱུར་ནས་ཉོན་མོངས་པའི་སྟོབས་ཀྱིས་རང་ཉིད་ལ་ཚུར་རང་དབང་མེད་པ་བཟོ་ཡོང་བ་རེད། དེའི་སྟོབས་ཀྱིས་ང་ཚོ་འཁོར་བ་ལ་སྡུག་བསྔལ་མྱོང་བ་རེད། ད་ལྟ་ཆོས་ཉམས་ལེན་བྱེད་ཟེར་དུས་དང་པོ་ཞེ་སྡང་ཡག་པོ་མིན་པ། ཁོང་ཁྲོ་དང་འདོད་ཆགས་སེར་སྣ་ཕྲག་དོག་སོགས་ཡག་པོ་མིན་པ་དང༌། སེམས་མ་སྐྱིད་པར་བཟོ་མཁན་གྱི་རྣམ་རྟོག་དེ་ལ་སྐྱོན་དུ་བལྟས་ནས་དེས་ང་འཁོར་བ་ཐོག་མེད་ནས་མ་སྐྱིད་པར་བཟོས། འཁོར་བ་ཐོག་མེད་ནས་ང་ལ་སྡུག་བསྔལ་བཟོས། འཁོར་བ་ཐོག་མེད་ནས་དཀའ་ངལ་བཟོས་པ་རེད། སྡུག་བསྔལ་དང་དཀའ་ངལ་བཟོ་མཁན་དེ་ལས་སུ་མ་རུང་བའི་རྣམ་རྟོག་དང་ཉོན་མོངས་པ་ཟེར་བ་དེ་འདི་རེད་འདུག །ད་ལན་དལ་འབྱོར་གྱི་མི་ལུས་ཐོབ། སངས་རྒྱས་ཀྱི་བསྟན་པ་མཇལ། དམ་པའི་ཆོས་ལ་ཐོས་བསམ་སྒོམ་པའི་ཉམས་ལེན་བྱེད་པའི་གོ་སྐབས་ཡོད་པའི་སྐབས་འདིར། ངས་དུས་ཚོདའཕྲོ་བརླག་མེད་པར་ཉོན་མོངས་དང་པོར་ངོས་ཟིན་ཐུབ་པ་ཞིག་ཅི་ནས་ཀྱང་བྱ་སྙམ་པའི་སྒོ་ནས། དྲན་པ། ཤེས་བཞིན། བག་ཡོད་པར་བསྟེན་པ་ཡིན་ན། དེའི་སྟོབས་ཀྱིས་ཉོན་མོངས་རྩད་ནས་བཅོམ་མ་ཐུབ་ནའང་ཉོན་མོངས་པས་རང་ལ་ཚུར་རང་དབང་མེད་པར་བཟོ་བའི་གོ་སྐབས་དེ་ཉུང་དུ་འགྲོ་བ་ལ་ཕན་ཐོགས་ཆེན་པོ་ཡོད། དཔེར་ན་རྐུན་མ་ཞིག་ཡོང་གི་འདུག་ཅེས་ལབ་ཡོང་དུས་རྐུན་མ་ཡོང་མ་ཡོང་དེ་སོ་སོས་བག་ཟོན་བྱེད། སྒོ་དང་སྒེ་ཁུང་གྲས་ལ་དུས་དང་རྣམ་པ་ཀུན་ཏུ་དྲན་ཤེས་བག་ཡོད་ཀྱི་སྒོ་ནས་བག་ཟོན་བྱེད་པ་ཡིན་ན། རྐུན་མ་ནང་ལ་འཛུལ་བའི་གོ་སྐབས་ཆེས་ཉུང་ཉུང་རེད། རང་ཉིད་སེམས་རྣམ་གཡེང་གི་དབང་དུ་སོང༌། དྲན་ཤེས་བག་ཡོད་མེད་པར་བྱེད་པ་ཡིན་ན། རྐུན་མ་ནང་ལ་འཛུལ་བའི་གོ་སྐབས་ཆེན་པོ་ཡོད། དཔེ་དེ་བཞིན་དུ་ཉོན་མོངས་པ་དེ་ཚུར་རང་ལ་རང་དབང་ཡོད་པར་བཟོ། ཉོན་མོངས་པ་དེ་ཚུར་རང་གི་དབང་དུ་སྡུད་པའི་གོ་སྐབས་བྱེད་པ་ཡིན་ན། ཆོས་ཀྱི་ཉམས་ལེན་འགོ་བཙུགས་པ་རེད། ང་ཚོས་ཆོས་བྱེད་ཅེས་པ་དེ་ཡང་ཆོས་གཞན་གང་ཡང་བྱེད་རྒྱུ་ཡོད་མ་རེད། སེམས་ལ་བཟོ་བཅོས་གཏོང་རྒྱུ་ཞིག་རེད། ཐོག་མེད་ནས་བྱས་ད་ལྟའི་བར་སེམས་བཟོ་བཅོས་གཏོང་མ་ཐུབ་ནས་ང་རང་ཚོ་དོན་མེད་པའི་སྡུག་བསྔལ་འདི་བྱུང་བ་ཡིན་དུས་རྣམ་གཡེང་མེད་པ་ཞེས་པའི་ཚིག་འདི་ནུས་པ་ཆེན་པོ་ཡོད། རྣམ་གཡེང་མེད་ན་དགེ་སྦྱོར་ངང་གིས་འཕེལ་ཞེས་རྣམ་གཡེང་མེད། རྣམ་གཡེང་སྐྱེ་བའི་རྐྱེན་མེད་པས་རྣམ་པར་གཡེང་བའི་ཡུལ་མེད། རྣམ་པར་གཡེང་བའི་ཡུལ་མེད་པས་རྣམ་གཡེང་གི་སྟོབས་དང་ནུས་པ་ཉུང་དུ་སོང༌། དེའི་སྟོབས་ཀྱིས་དེ་དང་འཛིན་སྟངས་དངོས་འགལ་དུ་གྱུར་པའི་ལུས་ངག་ཡིད་གསུམ་གྱི་དགེ་བ་དེ་དག་གོང་ནས་གོང་དུ་འཕེལ་ཡོང་བ་རེད། ངང་གིས་འཕེལ་གསུངས་པ་དེ་རང་བཞིན་གྱིས་འཕེལ་གྱི་རེད་གསུངས་པ་རེད། འོ་ན་སྔར་ཅིའི་ཕྱིར་འཕེལ་མི་ཐུབ་ཅེ་ན། ཆགས་སོགས་ཉོན་མོངས་པའི་རྣམ་པར་རྟོག་པས་དགེ་སྦྱོར་འཕེལ་མི་བཅུག་པ་རེད། ད་གཟོད་རྣམ་གཡེང་ཉུང་དུ་གཏང་ན། དགེ་སྦྱོར་གོང་ནས་གོང་དུ་འཕེལ་བའི་གོ་སྐབས་ཡོད། དེའི་ཕྱིར་ན་རྣམ་གཡེང་མེད་པས་དགེ་སྦྱོར་ངང་གིས་འཕེལ། རྐྱེན་ཡུལ་ངན་སྤངས༌། རྣམ་གཡེང་དང་བྲལ། ཆགས་སོགས་དང་བྱུར་དང་མི་མཐུན་པ་དང་བར་ཆད་ལ་སོགས་དང་བྲལ་བའི་ཤུལ་ལ་རིག་པ་དྭངས། རིག་པ་དྭངས་པས་ཐོས་བསམ་སྒོམ་པའི་ཤེས་རབ་སོགས་མ་སྐྱེ་བ་ཐམས་ཅད་སྐྱེ། སྐྱེས་པ་ཐམས་ཅད་གོང་ནས་གོང་དུ་འཕེལ་བར་བརྟེན་རིག་པ་དྭངས་པས་ཆོས་ལ་ངེས་ཤེས་སྐྱེ་ཞེས་གསུངས། ཆོས་ལ་ངེས་ཤེས་སྐྱེ་ཞེས་པ་སྟེ་ཆོས་ལ་སྤྲོ་བ་ལྡན་པ་ཞིག་ལ་གོ་བ་རེད། དགེ་བ་ཅི་འདྲ་ཞིག་སྒྲུབ་ན་ཡང༌། དེ་ལ་དགོས་པ་རྒྱུ་མཚན་འདི་འདྲ་ཞིག་འདུག །ཆོས་འདི་སྒྲུབ་ན་ཕན་ཡོན་དང་ཡོན་ཏན་འདི་འདྲ་ཞིག་འདུག །མ་བསྒྲུབས་ན་སྐྱོན་དང་ཉེས་དམིགས་འདི་འདྲ་ཞིག་འདུག་ཅེས་སྐྱོན་དང་ཡོན་ཏན་སོ་སོར་ཕྱེ་ཤེས་ཡོང་དུས། སྐྱོན་དེ་དོར་བ་ལ་སྤྲོ་བ་ཡོད་ལ་དཀའ་ངལ་མེད་པ། ཆོས་སྒྲུབ་པ་ལ་སྤྲོ་བ་ཡོད་ལ་དཀའ་ངལ་མེད་པ་རེད། བརྩོན་འགྲུས་ཟེར་ནའང་འདི་རེད། སྤྲོ་བ་དང་བཅས་པའི་ཐོག་ནས་ཆོས་སྒྲུབ་ཡོང་དུས་དཀའ་ཚེགས་འཚེར་སྣང་དང་མི་འདོད་པ་སོགས་མི་ཡོང་བ་རེད། དཀའ་ཚེགས་དང་འཚེར་སྣང་དེ་ཡང་སྤྲོ་བ་མེད་ནས་ཡོང་བ་རེད། དེའི་ཕྱིར་ན་རིག་པ་དྭངས་པས་ཞེས་རིག་པ་དྭངས་ཡོང་དུས་ཐོས་པའི་ཤེས་རབ། བསམ་པའི་ཤེས་རབ། སྒོམ་པའི་ཤེས་རབ་སྐྱེ་ཡི་རེད། ཐོས་བསམ་སྒོམ་གསུམ་གྱི་ཤེས་རབ་སྐྱེས་ན། དེའི་སྟོབས་ཀྱིས་ཆོས་ལ་ངེས་ཤེས་སྐྱེ། ངེས་ཤེས་ཞེས་པ་དེ་འདིས་ཕན་གྱི་འདུག །འདི་ལ་དགོས་པ་འདུག་བསམ་པའི་གདེང་ཞིག་ཐོབ་དགོས་པ་རེད། དེའི་ཕྱིར་ན་དབེན་པ་བསྟེན་པ་རྒྱལ་སྲས་ལག་ལེན་ཡིན་ཞེས་དབེན་པ་བསྟེན་དགོས་གསུངས། རྒྱུ་མཚན་དེ་འདྲ་ཞིག་མ་ཤེས་པར་དེ་ག་ཙམ་ཉིན་བཅོ་ལྔ་ཉི་ཤུའི་ལམ་ལ་ཡར་ཕྱིན་ནས་མི་གཏན་ནས་མེད་པའི་ས་ཆ་ལ་མཚམས་བསྡད་ན་ཡང༌། སེམས་འདི་ཉོན་མོངས་པ་ལས་ཐག་སྲིང་མ་ཐུབ། ཡུལ་ངན་རྣམ་པར་གཡེང་བ་སོགས་སྤངས་མ་ཐུབ་ན། ལུས་རི་ཁྲོད་ལ་ཡོད་ཀྱང་སེམས་རི་ཁྲོད་ལ་མེད། སེམས་དེ་ད་དུང་ཡང་ཆགས་སོགས་ཉོན་མོངས་པའི་རྗེས་སུ་བཏང་ནས་གཟུགས་གཅིག་པུ་ཡར་བྲག་ཕུག་ནང་ལ་གབ་བསྡད་པས་མི་ཕན། སྤྱོད་འཇུག་བསམ་གཏན་ལེའུ་ནང་ལུས་རྣམ་པར་གཡེང་བ་སྤང་བ་དང་སེམས་རྣམ་པར་གཡེང་བ་སྤང་པ་གཉིས་ཀྱི་ནང་ཏིང་ངེ་འཛིན་གྱི་ལམ་གྱི་རིམ་པ་མཐའ་དག་བསྒྲིགས་ནས་གསུངས་ཡོད། དེ་འདྲ་ཡིན་དུས་དབེན་པ་བསྟེན་གསུངས།
འདིར་དབེན་པ་བསྟེན་ཞིང་སྒྲུབ་པ་ལ་སྙིང་པོར་བྱེད་པ་རྣམས་ཀྱིས།
ཞེས་སྙིང་པོར་བྱེད་པ་ཞེས་པ་སྟེ་དོན་གཙོ་བོར་བྱེད་པ། མཐར་ཐུག་པར་བྱེད་པ། གལ་ཆེན་པོར་བརྩི་བ་ལ་སྙིང་པོར་བྱེད་པ་ལ་སྒྲུབ་པ་སྙིང་པོར་བྱེད་པ་གསུངས། འོ་ན་སྒྲུབ་པ་སྙིང་པོར་བྱེད་པ་ལ་ངེས་པར་དུ་རི་ཁྲོད་ལ་འགྲོ་དགོས་སམ་ཞེ་ན། མི་དགོས་ཏེ། ང་རང་ཚོ་ཉིན་རེའི་ནང་ཐོས་བསམ་སྒོམ་གསུམ་གྱི་ཉམས་ལེན་དང་ལུས་ངག་ཡིད་གསུམ་གྱི་དགེ་བ་སོགས་གང་དང་གང་སྒྲུབ་པ་དེ་དངོས་དང་བརྒྱུད་ནས་རྣམ་རྟོག་དང་བདག་འཛིན་ཉོན་མོངས་པའི་གཉེན་པོ་ལ་འགྲོ་ཐུབ་ན། ངེས་པར་དུ་བྲག་ཕུག་ནང་དང་ས་ཐག་རིང་པོ་ལ་ཞིག་ལ་འགྲོ་དགོས་དོན་མེད། ཆོས་ཉམས་ལེན་གང་དང་གང་བྱེད་པ་དེ་ཉོན་མོངས་པའི་གཉེན་པོ་ལ་འགྲོ་ན། ཆོས་སྒྲུབ་པ་སྙིང་པོར་བྱེད་པ་ཞེས་གསུངས་པའོ། །

ཞེས་རྣམ་སྣང་ཆོས་བདུན་བཅའ་བའི་ཐོག་ནས། ཀྱེ་མ། དལ་འབྱོར་མི་ལུས་འདི་ནི་ཤིན་ཏུ་རྙེད་དཀའ་ཡང་ད་ལན་ལན་ཅིག་རྙེད་ཡོད། ལན་ཅིག་རྙེད་པའི་སྐབས་སུ་དལ་འབྱོར་མི་ལུས་ཐོབ་ཡོད་ཅེས་བློ་བདེ་བྱས་པ་ལ་དགོས་པ་དང་ཕན་ཐོགས་མེད་དེ། ང་ཚོ་མི་རྟག་པ་ཡིན་པའི་ཕྱིར་ན་དུས་ད་ལྟ་འཆི་བ་འོང་ངེས་པ་མེད། དེ་རིང་ཕྱི་དྲོ་ཤི་ཡོང་ངེས་པ་མེད། སང་ཉིན་ཤི་ཡོང་ངེས་པ་མེད་པས་མི་ལུས་རིན་ཆེན་རྙེད་པའི་རྙེད་ཐོག་ནས་ཆོས་ད་ལྟ་ཉིད་ནས་སྒྲུབ་དགོས། ཆོས་ད་ལྟ་ཉིད་ནས་མ་བསྒྲུབས་པར་ཆོས་སང་ཉིན་སྒྲུབ། གནངས་ཉིན་སྒྲུབ་བོ་སྙམ་ན། མི་རྟག་པ་རེད། འཆི་བར་ངེས་པ་རེད། དེ་ལ་བརྟེན་ནས།
ཚེ་འདི་མཛའ་བཤེས་ཚོགས་འདུས་མི་བཞིན་འགྱེས། །

ཀྱེ་མ་འདུས་བྱས་རྣམས་མི་རྟག །

གསུངས་པ་བཞིན་འདུས་བྱས་ཀྱི་ཆོས་རྣམས་སྐད་ཅིག་སྐད་ཅིག་གིས་སྐྱེ་འགག་བྱེད་པ་ཅན་ཞིག་རེད་མ་གཏོགས་རང་གི་རང་བཞིན་དེ་རྟག་ཏུ་མི་གནས་པ་རེད། ཁོ་རྒྱུ་རྐྱེན་ལ་ལྟོས་པ་ཡིན་དུས་གང་རྒྱུ་རྐྱེན་ལ་ལྟོས་ན། དེ་ལ་རང་དབང་མེད། དེ་སྐད་ཅིག་གིས་འཇིག་པར་བྱེད། དེ་མི་རྟག་པ་རེད། དེ་རིང་མཉམ་དུ་ཡོད་ཀྱང་སང་ཉིན་ལྷན་དུ་འཛོམ་དུ་མེད་པ། ལོ་གཉིས་གསུམ་མཉམ་དུ་གནས་ན་ཡང་ལོ་གཉིས་གསུམ་གྱི་རྗེས་ཡང་ན་རང་ཉིད་འཆི་བ་རེད། ཡང་ན་གཞན་འཆི་བ་རེད། དུས་རྟག་ཏུ་མཉམ་དུ་སྡོད་རྒྱུ་མེད། དེའི་ཕྱིར་ན་དེ་དག་གི་དོན་ལ་ཕན་ཚུན་གཅིག་གིས་གཅིག་ལ་འཁྲུག་རྩོད་རྒྱག་པ་དང་ངན་པར་སྨྲ་བ། ལག་འཛིང་རྒྱག་པ་སོགས་དང༌། མི་མཐུན་པའི་གཅིག་གིས་གཅིག་ལ་མ་བཟོད་པའི་ཕྲག་དོག །ཁོན་འཛིན། སྙིང་ནད་ཟིན་པ་སོགས་ལ་ཡང་དགོས་པ་མེད་དེ། ཚེ་ལ་ངེས་པ་ནི་མེད། ཁོང་ཚོ་ཡང་འཆི་འགྲོ་ལ་རང་ཉིད་ཀྱང་འཆི་འགྲོ་བ་རེད། དཔེར་ན་གྲོང་ཁྱེར་ནང་ཚོགས་འདུ་འཛོམས་སྐབས་སུ་དང་པོ་འཛོམས་པ་རེད། འོན་ཀྱང་ཚོགས་འདུའི་བྱ་བ་གྲུབ་ན་སོ་སོར་གྱེས་ཏེ་རང་རང་གི་ས་ལ་འགྲོ་བ་མ་གཏོགས་རྟག་ཏུ་མཉམ་དུ་མི་གནས་པ་བཞིན་ནོ། །དེའི་ཕྱིར་ན་གང་མཉམ་དུ་ཡོད་པའི་རིང་ལ་ཁ་ཤགས། འཐབ་མོ། རྩོད་པ། འཁྲུག་པ། ནང་ཕན་ཚུན་ལ་སྡང་ཟུག །ཁོན་འཛིན་སོགས་མ་བྱས། དེ་བྱེད་པ་ལ་དགོས་པ་ཡང་མེད་དེ། དེས་རང་གི་ལས་ངན་པ་བསགས་པ་ཙམ་གྱི་འབྲས་བུ་ལས་གཞན་ཕན་ཐོགས་ཅི་ཡང་མེད། ད་བཞིན་དུ་རང་ལ་དེ་རང་ཡོད་དགོས། དེ་མེད་ན་ཐབས་ཆག་སྙམ་པའི་སེམས་ཤིན་ཏུ་ཞེན་ཅིང་ཆགས་པའི་ཞེན་ཆགས་ཀྱང་བྱེད་རྒྱུ་མེད་དེ། དེ་ལྟར་བགྱིས་པ་ཡིན་ན། སོ་སོའི་སེམས་ལ་སྡུག་བསྔལ་ཆེན་པོ་འབྱུང༌། དགའ་བ་དང་སྡུག་པ་གཉིས་ཀ་ཡང་མེད་པ། ཐམས་ཅད་མི་རྟག་པའོ་སྙམ་པའི་སྒོ་ནས་ད་ལྟ་ཉིད་ནས་སེམས་ལ་མི་རྟག་པའི་འདུ་ཤེས་ཡང་ནས་ཡང་དུ་སྐྱེད་ཐུབ་ན། ཕན་ཐོགས་ཆེན་པོ་ཡོད། བཅོམ་ལྡན་འདས་ཀྱིས་ཀྱང་འདུ་ཤེས་ཐམས་ཅད་ཀྱི་ནང་ནས་མི་རྟག་པའི་འདུ་ཤེས་མཆོག་ཅེས་གསུངས་པ་རེད། ཆོས་སྒྲུབ་པ་ལ་ཡང་མི་རྟག་པ་ཞེས་པ་དེ་ཕན་ཐོགས་ཆེན་པོ་ཡོད། ཆོས་སང་ཉིན་སྒྲུབ་གནངས་ཉིན་སྒྲུབ་བོ་སྙམ་པའི་དོན་གང་ཡིན་ཞེ་ན། མི་རྟག་པའི་འདུ་ཤེས་མེད་པའི་ཉེར་ལེན་རེད། ད་ལྟ་ལུས་བདེ་པོ་འདུག །བདུན་ཕྲག་རྗེས་མ་ཡིན། ཟླ་བ་རྗེས་མ་ཡིན་སྙམ་པ་ལྟ་བུ་མིན་པར། འཆི་བ་ལ་ངེས་པ་རེད། མ་ཤི་བ་ལ་འགྲོ་ས་མེད། འཆི་བ་ལ་ངེས་པར་དུ་ན་སྟེ་འཆི་དགོས་པའི་ཁྱབ་པའང་མེད་པས་ཚུལ་ཇི་ལྟར་འཆི་འོང་ངེས་གཏན་མེད། འཆི་བའི་རྐྱེན་ནི་མང༌། གསོན་པའི་རྐྱེན་ནི་ཉུང༌། སྐབས་འགར་གསོན་པའི་རྐྱེན་དེ་ཡང་ཤི་བའི་རྐྱེན་དུ་འགྲོ། མཚམས་རེ་ནི་གློ་བུར་དུ་ཡས་ནས་མར་རྡོ་སོགས་ཟག་སྟེ་མགོ་ལ་ཕོག་ནས་ཤི་བ་སོགས་གློ་བུར་གྱི་རྐྱེན་ལ་བརྟེན་ནས་འཆི་ངེས་པའང་མང་བས་ཚུལ་གང་ལ་བསམ་ཡང་ངེས་པ་མེད་པའི་ཕྱིར། དུས་ད་ལྟ་ཆུ་ཚོད་གཅིག་ཡིན་ནའང་འདྲ། ཆུ་ཚོད་ཕྱེད་ཀའི་རིང་གི་ཚེ་ན་འདྲ། སྐར་མ་བཅོ་ལྔ་ལས་མེད་ན་ཡང་ཁྱད་མེད། གང་ལྟར་ད་ལྟ་ཚེ་ཡོད་པའི་སྐབས་འདིར་ངས་ཆོས་སྒྲུབ་དགོས་ཀྱི་འདུག །ཆོས་ད་ལྟ་ཉིད་ནས་སྒྲུབ་དགོས་ཀྱི་འདུག་སྙམ་སྟེ་རང་གིས་རང་ལ་ཆོས་ལ་འཇུག་པའི་སྐུལ་མ་བྱེད། ཆོས་ལ་སྐུལ་མ་བྱེད་པ་ལ་མི་རྟག་པ་འདི་མཆོག་ཏུ་གྱུར་པ་ཡིན། དེ་ལ་བརྟེན་ནས་བཀའ་གདམས་པའི་བླ་མ་གོང་མ་ཐམས་ཅད་ཀྱིས་མི་རྟག་པའི་ཉམས་ལེན་གནང་བ་རེད། མི་རྟག་པའི་འདུ་ཤེས་ལ་བརྟེན་ནས་དང་པོ་ཆོས་ལ་འཇུག །མི་རྟག་པའི་འདུ་ཤེས་ལ་བརྟེན་ནས་ཆོས་དངོས་གཞིའི་ཉམས་ལེན་གནང༌། མི་རྟག་པའི་འདུ་ཤེས་ལ་བརྟེན་ནས་ཆོས་མཐར་ཕྱིན་པ་སོགས་དགོས་པ་དང་ཕན་ཆེན་པོ་ཡོད་པར་གསུངས། མགོན་པོ་ཀླུ་སྒྲུབ་ཀྱིས་ཀྱང༌།
སྐྱེ་བ་ཤེས་པས་འཇིག་པ་ཤེས། །
འཇིག་པ་ཤེས་པས་མི་རྟག་ཤེས། །
མི་རྟག་ཉིད་ལ་འཇུག་ཤེས་པས། །
དམ་པའི་ཆོས་ནི་རྟོགས་པར་འགྱུར། །
ཞེས་པ་སྟེ་ཐོག་མར་སྐྱེ་བ་ཤེས་དགོས། སྐྱེ་བའང་མི་རྟག་པ་ལ་བརྟེན་ནས་ཡོང་བ་ཡིན། སྐྱེ་བ་ཤེས་པས་འཇིག་པ་ཤེས་ཞེས་སྐྱེ་བ་ཤེས་པས་འཇིག་པའམ་ཤི་བ་ཤེས་པ། འཇིག་པ་ཤེས་པས་མི་རྟག་ཤེས་ཞེས་སྐྱེ་འཇིག་གི་རྒྱུ་མཚན་ཤེས་པས་མི་རྟག་པ་ལ་ངེས་ཤེས་དྲངས་འོང་བ་ཡིན། མི་རྟག་ཉིད་ལ་འཇུག་ཤེས་པས། །དམ་པའི་ཆོས་ནི་རྟོགས་པར་འགྱུར། །ཞེས་མི་རྟག་པའི་འདུ་ཤེས་ལ་བརྟེན་ཏེ་རིམ་གྱིས་དམ་པའི་ཆོས་ཞེས་པ་སྟེ་ཟབ་མོ་སྟོང་པ་ཉིད་ཀྱི་གནས་ལུགས་སྤྲོས་པ་ཐམས་ཅད་བྲལ་བའི་གནས་ལུགས་དེ་དེའི་སྟེང་ནས་རྟོགས་ཐུབ་ཀྱི་རེད། འདི་སྔོན་དུ་འགྲོ་བའི་ཆོས་དེ་གང་ཡིན་ཞེ་ན། སྐྱེ་འཇིག་གི་རང་བཞིན་ཅན་གྱི་མི་རྟག་པ་ལ་བློ་ལེགས་པར་གོམས་དགོས་པ་ཡིན། དེ་གོམས་ན་དེའི་འབྲས་བུ་རབ་ཏུ་མི་གནས་པའི་གནས་ལུགས་དེ་མཐོང་བ་ལ་དཀའ་ཚེགས་མེད་ཅེས་གསུངས་པའི་ལམ་གྱི་རིམ་པ་དེ་ལྟར་བགྲོད་དགོས་པ་ཡིན། དེའི་ཕྱིར་ན་ཆོས་བྱེད་པ་ཀུན་གྱིས་མི་རྟག་པའི་འདུ་ཤེས་ཞེས་པ་ལ་བསམ་མནོ་གཏོང་རྒྱུ་གལ་ཆེ། ཚེ་འདིའི་གཏམ་སྙན་གྲགས་ཐོབ་གོ་གནས་སོགས་ཡོང་སྙམ་པའི་རེ་བ་མང་པོ་ཞིག་བཅོལ། གྲ་སྒྲིག་མང་པོ་ཞིག་བྱེད་ཀྱི་ཡོད། རེ་བ་བཞིན་འོང་དཀའ། བདུན་ཕྲག་རྗེས་མར་ངས་དེ་འདྲ་ཞིག་བྱེད། ཟླ་བ་རྗེས་མ་ལ་ངས་དེ་འདྲ་ཞིགབྱེད་སྙམ་པའི་སེམས་ནང་འཆར་གཞི་བང་རིམ་མང་པོ་བསྒྲིགས་ནས་བཞག་ཡོད། ཡིན་ན་ཡང་འཆར་གཞི་དེ་དང་རང་གི་འདོད་པ་དེ་ཇི་བཞིན་དུ་སྒྲུབ་ཐུབ་པ་ཁག་པོ་རེད། སྔོན་ལ་མི་རྟག་པ་སླེབ་ཡོང་ངམ་འཆར་གཞི་དེ་སྔོན་དུ་སླེབ་ཡོང་ངེས་པ་མེད་པས་མི་རྟག་པའི་འདུ་ཤེས་བསྐྱེད་ན། ཆོས་ལ་ཕན་ཐོགས་ཆེན་པོ་ཡོད་ཅེས་གསུངས། སྔོན་བླ་མ་གོང་མ་རྣམས་ཀྱི་ཐུགས་དམ་ཉམས་བཞེས་གཙོ་བོ་མི་རྟག་པ་རེད། ང་ལ་དུས་ཚོད་མི་འདུག །ནམ་འཆི་འོང་ངེས་པ་མེད། འཇིག་རྟེན་པའི་ཆགས་སྡང་གི་གཏམ་འཆད་པ་ལ་ལོང་མེད། གང་ཡིན་ཞེ་ན། ཆོས་སྒྲུབ་དགོས་ཀྱི་འདུག །སང་ཉིན་འཆི་བའི་སྐབས་སུ་ཆོས་མ་གཏོགས་གོ་གནས་ཆེ་བས་ཕན་གྱི་མི་འདུག །དཔལ་འབྱོར་ཕྱུག་ཀྱང་ཕན་གྱི་མི་འདུག །སྐད་གྲགས་ཆེ་ཡང་ཕན་གྱི་མི་འདུག །དེ་དག་ཐམས་ཅད་རྗེས་སུ་གཞག་རྒྱུ་རེད་མ་གཏོགས་རང་དང་མཉམ་དུ་མིང་དང་སྙན་གྲགས། དཔལ་འབྱོར་དང་ལོངས་སྤྱོད་སོགས་མཉམ་དུ་ཁྱེར་བར་རང་ལ་རང་དབང་མི་འདུག །དེའི་ཕྱིར་ད་ལྟ་ཆོས་སྒྲུབ་པའི་སྐབས་ཉིད་ནས་མི་རྟག་པ་རེད། ཚེ་ལ་སྙིང་པོ་ཡོད་མ་རེད། ནོར་ལོངས་སྤྱོད་གྲས་དགོས་པ་ཡོད་མ་རེད། འཛམ་བུ་གླིང་ལ་དབང་བའི་རྒྱལ་པོ་ལྟ་བུ་ཞིག་ཡིན་རུང་འཆི་བའི་ཚེ་ཕན་པ་མི་འདུག ཞེས་པ་སྟེ་ཤི་ཚར་ན་རང་དབང་མེད་དེ། གསོན་པའི་དུས་ན་ཆོས་མ་བསྒྲུབས། དེའི་ཕྱིར་ཤི་ཚར་དུས་ཙ་ན་ལས་ཀྱི་སྟོབས་ཀྱིས་ལས་དགེ་བ་བསགས་ཡོད་ན་དེའི་རྗེས་སུ་འགྲོ་བ། ལས་མི་དགེ་བ་བསགས་ཡོད་ན་མི་དགེ་བའི་སྟོབས་ཀྱིས་རྗེས་སུ་འབྲང་བ་ཁོ་ན་ལས་ང་མཐོ་རིས་གནས་སུ་འགྲོ་རྒྱུར་གདམ་ག་དང༌། ངན་སོང་གནས་རྣམས་སུ་མི་འགྲོ་ཞེས་སོ་སོས་ཐག་བཅད་དེ་གདམ་ག་རྒྱག་ཏུ་མེད། དམྱལ་བ་ཡི་དྭགས་དུད་འགྲོ་རྣམས་སུ་འགྲོ་བྱེད་ཀྱི་རྒྱུ་བསགས་ན་འབྲས་བུ་སྡུག་བསྔལ་འབྱུང་བ་རྟེན་འབྲེལ་ཡིན། དེའི་ཕྱིར་ན་གདམ་ག་མ་ནོར་བར་བྱས་ཏེ་ད་ལྟ་ཉིད་ནས་ལས་མི་དགེ་བ་སྤང༌། དགེ་བ་སྒྲུབ་དགོས་པ་ཡིན། འོ་ན་མི་དགེ་བ་སྤངས་པའི་ཁེ་ཕན་གང་ཡོད་ཅེ་ན། ངན་སོང་ལ་འགྲོ་མི་དགོས་པ་ཡིན། དགེ་བ་སྒྲུབ་པའི་ཁེ་ཕན་གང་ཡོད་ཅེ་ན། མཐོ་རིས་ལྷ་དང་མིའི་བདེ་འགྲོ་འཐོབ་ཀྱི་རེད། དེས་ན་ཆོས་སྒྲུབ་པའི་ཚེ་མི་དགེ་བ་སྦོང་བ་དང་དགེ་བ་སྒྲུབ་པའི་ཐབས་ཤེས་བྱེད་དགོས་པའོ། །

ཞེས་འོ་ན་དལ་འབྱོར་རྙེད་དཀའ་བ་ལ་བསམ་ནས་དགེ་བ་སྒྲུབ་དགོས་པ། འཆི་བ་མི་རྟག་པ་ལ་བསམ་ནས་ཆོས་སྒྲུབ་དགོས་པ། ལས་རྒྱུ་འབྲས་ལ་བསམ་ནས་ཆོས་སྒྲུབ་དགོས་པ། དལ་འབྱོར་རྙེད་དཀའ་བ། འཆི་བ་མི་རྟག་པ། ལས་རྒྱུ་འབྲས་གསུམ་ལ་བརྟེན་ནས་ང་ཚོས་བློ་གང་སྐྱེ། སུ་ལ་རེ་དགོས་ཤེ་ན། དེ་བས་འཁོར་བ་འདི་ལ་སྐྱོ་ཤས་སྐྱེས། ཞེས་འཁོར་བའི་ཉེས་དམིགས་ཏེ་བདེ་འགྲོ་ལྷ་དང་མི། ངན་འགྲོ་དམྱལ་བ་སོགས་འགྲོ་བ་རིགས་དྲུག །ཁམས་གསུམ། གནས་རྒྱུད་ལམ་རྒྱུད། སྐྱེ་གནས་བཞི་སོགས་གང་དུ་ཕྱིན་ནའང༌། འཁོར་བ་ཁབ་ཀྱི་རྩེ་མོན། བདེ་བ་ནམ་ཡང་ཡོད་མ་ཡིན། ཞེས་གསུངས་པ་བཞིན་སྡུག་བསྔལ་ཁོ་ན་ལས་བདེ་བ་ནམ་ཡང་མེད། སྡུག་བསྔལ་མ་གཏོགས་མེད་པ་ཡིན། འཁོར་བ་ཁབ་ཀྱི་རྩེ་མོ་བདེ་བ་ནམ་ཡང་ཡོད་མ་ཡིན་གསུངས་པའོ། །སྡུག་བསྔལ་མ་གཏོགས་བདེ་བ་ནམ་ཡང་མེད་པ་ཡིན། སྡུག་བསྔལ་གྱི་རང་བཞིན་རེད། སྡུག་བསྔལ་གྱི་སྡུག་བསྔལ། འགྱུར་བའི་སྡུག་བསྔལ་གཉིས། གཉིས་པོའི་འབྱུང་ཁུངས་རྩ་བ་དེ་ཁྱབ་པ་འདུ་བྱེད་ཀྱི་སྡུག་བསྔལ་རེད། ཉེར་ལེན་གྱི་ཕུང་པོ་ལྔ་པོ་འདི་བླངས་ཙམ་གྱིས་སྡུག་བསྔལ་ཕྱི་མ་གཉིས་དེ་འདྲེན་ཡོང་བ་རེད། དེ་ལ་བརྟེན་ནས་ཁྱབ་པ་འདུ་བྱེད་ཀྱི་སྡུག་བསྔལ་དེ་སོ་སོ་སྐྱེ་བོས་སྡུག་བསྔལ་དུ་མི་ཚོར་ཡང་འཕགས་པ་རྣམས་ཀྱིས་ནི་སྡུག་བསྔལ་དུ་གཟིགས་པ་ཡིན། སོ་སོ་སྐྱེ་བོ་རྣམས་ཀྱིས་ཁྱབ་པ་འདུ་བྱེད་ཀྱི་སྡུག་བསྔལ་ལས་སྐྱེས་པའི་སྡུག་བསྔལ་གྱི་སྡུག་བསྔལ་དང༌། འགྱུར་བའི་སྡུག་བསྔལ་དེ་སྡུག་བསྔལ་དུ་མཐོང་གི་ཡོད། དེའི་ཕྱིར་ན་འཁོར་བ་འདི་ལ་སྐྱོ་ཤས་སྐྱེས་ཞེས་གསུངས། སྐྱོ་ཤས་ཞེས་པ་ནི་ངེས་འབྱུང་ལ་གོ་བ་ཡིན། བཅོམ་ལྡན་འདས་ཀྱིས་དང་པོ་ངེས་པར་འབྱུང་བ་ཚུལ་ཁྲིམས་ཀྱི་དབང་དུ་བྱས་ཏེ་ཞེས་ངེས་འབྱུང་གི་ཚུལ་ཁྲིམས་ཞེས་པའང་དེ་ཡིན། བདེན་པ་བཞིའི་ཉམས་ལེན་ཡང་དེ་ནས་དབུ་བཙུགས་དགོས། སྐྱབས་འགྲོའི་ཉམས་ལེན་ཡང་དེ་ནས་འགོ་བཙུགས་དགོས་པས་འཁོར་བ་ལ་སྐྱོ་ཤས་མེད། དལ་འབྱོར་རྙེད་དཀའ་བ། འཆི་བ་མི་རྟག་པ། ལས་རྒྱུ་འབྲས། འཁོར་བའི་ཉེས་དམིགས་བཞི་ལ་བསམ་མནོ་མ་བཏང་ན། སྐྱབས་འགྲོའི་སྡོམ་པ་ཞུས་པས་མི་ཕན། ཀྱེ། གཙུག་ཕུད་འབྲེག་པར་སྤྲོའམ། སྤྲོ་ལགས། ཞེས་ཟེར་བ་ཙམ་མ་གཏོགས་སྐྱབས་འགྲོ་རྣམ་དག་རྒྱུད་ལ་སྐྱེ་མི་ཐུབ་པ་རེད། སྐྱབས་འགྲོ་རྣམ་པར་དག་པ་རྒྱུད་ལ་སྐྱེ་བ་ལ་སྔོན་འགྲོའི་ཆོས་བཞིའི་བློས་དྲངས་ཏེ་སྐྱབས་འགྲོ་རྣམ་པར་དག་པ་རྒྱུད་ལ་སྐྱེ་དགོས། དེའི་ཕྱིར་ན་འཁོར་བ་འདི་ལ་སྐྱོ་ཤས་སྐྱེས་ཞེས་གསུངས་པའོ། །

ཅེས་དུས་ཀྱི་ཐོག་ནས་བསམ་ན། བྱང་ཆུབ་སྒྲུབ་པའི་དུས་ཚོད་དེ་ད་རེས་རེད་གསུངས་པ་ཡིན། དལ་འབྱོར་གྱི་མི་ལུས་ཐོབ་བཞིན་པ། རྐྱེན་དགེ་བའི་བཤེས་གཉེན་དང་མཇལ་བཞིན་པ། གདམས་ངག་ཉམས་ལེན་བྱེད་པའི་དུས་ཚོད་དང་གོ་སྐབས་ཡོད་བཞིན་པའི་སྐབས་འདིར་དུས་ཚོད་འཕྲོ་བརླག་ཏུ་མ་བཏང་བར་ཆོས་སྒྲུབ་པའི་དུས་ཚོད་དེ་ད་རེས་ཡིན་ཞེས་གསུངས་པའོ། །
ཚེ་ལ་ལོང་མེད་སྒྲུབ་ལ་བརྩོན་འགྲུས་བྱ། །
ཞེས་པ་སྟེ་ཚེ་ལ་ལོང་མེད་ཅེས་གསུངས། དེ་ཡང་སང་ཉིན་ཆོས་སྒྲུབ་ཀྱི་ཡིན། གནངས་ཉིན་ཆོས་སྒྲུབ་ཀྱི་ཡིན། ལོ་གཞུག་མ་ལོ་གསུམ་ཕྱོགས་གསུམ་འཛུལ་འགྲོ་ཡི་ཡིན། ལོ་བཞིའི་རྗེས་ལོ་གསུམ་ཕྱོགས་གསུམ་འཛུལ་འགྲོ་ཡི་ཡིན་ཞེས་བརྗོད་ཀྱི་ཡོད་ཀྱང༌། ལོ་བཞིའི་མཇུག་ཏུ་རང་ཉིད་བསྡད་ཐུབ་པའི་རང་ལ་རང་དབང་ཡོད། སོ་སོའི་སེམས་ལ་ང་ལོ་བཞིའི་མཚམས་ལ་ཏན་ཏན་སྡོད་ཀྱི་རེད་བསམ་པའི་ཡིད་ཆེས་དང༌། བློ་གདེང་འཁེལ་བ་ཞིག་ཡོད་ན་ནི་གནས་ཚུལ་ཐ་དད་རེད། དེ་འགྲིག་གི་འདུག །དེ་ལྟར་མ་ཡིན་པར་རང་ཉིད་མི་རྟག་པའི་དབང་དུ་ཡོད་ན་ནི་ངེས་པ་མེད་པ་ཡིན། དེའི་ཕྱིར་ན་ཆོས་ད་ལྟ་དྲན་གྱི་ཡོད་ན། ད་ལྟ་ཉིད་ནས་སྒྲུབ་དགོས་པ་མ་གཏོགས་གཞུག་མ་དང་རྗེས་མ་སྒྲུབ་བོ་ཞེ་ན། ཕྱིས་དྲགས་པས་ཡོང་ཐབས་མེད་པ་ཡིན། འགག་ཅིག་ལ་ཐུག་ཡོང་དུས་སེམས་པ་སྐྱོ་བོ། མིག་ནས་མཆི་མ་འཁྲུག །ངས་ཆོས་སྒྲུབ་རྒྱུ་བྱུང་མ་སོང༌། སྒྲུབ་པའི་དུས་ལ་ནི་རྣམ་གཡེང་གི་དབང་དུ་སོང་སྟེ་སྒྲུབ་རྒྱུ་བྱུང་མ་སོང་སྙམ་སྟེ་ལག་པས་བྲང་ལ་སྤར་མོ་བཏབ། ད་ལྟ་དུས་ཚོད་དེ་འདྲ་ཞིག་ལ་ཐུག་དུས་ཆོས་སྒྲུབ་དགོས་བསམ་ཡང་དུས་མ་ཡིན་པ་རེད་བཞག །ད་སྒྲུབ་ལོང་མེད་པས་ཚར་བ་རེད་བསམ་ནས་སྔར་བསགས་པའི་ལས་ནག་པོ་དེ་དག་དྲན་ཏེ་འགྱོད་པ་དྲག་པོ་སྐྱེས། སེམས་ལ་སྡུག་བསྔལ་གྱིས་མནར། སྔར་དགེ་བ་བསྒྲུབས་པའི་བསམ་མནོ་ཟུར་ཙམ་ཡང་མེད། མི་འདོད་བཞིན་དུ་ངན་སོང་རྣམས་སུ་སྡུག་བསྔལ་བཟོད་དུ་མེད་པར་མྱོང་དགོས་པ་ཡིན། མགོ་རྙོག་ཚ་པོ་རེད། ད་ལྟའི་མགོ་རྙོག་ཚ་པོ་མིན་པར་རང་ཤི་བའི་སྐབས་དྭགས་པོ་རིན་པོ་ཆེས་བློ་བདེ་བག་ཡངས་ཀྱི་ངང་ནས་འགྲོ་བར་བྱའོ་ཞེས་གསུངས་པ་བཞིན་དགོས་པ་ཡིན། ཤི་བའི་སྐབས་སུ་བློ་བདེ་པོ་དང་བག་ཡངས་པོའི་ཐོག་ནས་འགྲོ་ཐུབ་མ་ཐུབ་དང་བློ་བདེ་པོ་ཡོང་མི་ཡོང་བཟོ་བའི་དུས་དེ་ད་ལྟ་རེད། རྒྱལ་བའི་བསྟན་པ་ལ་འཇུག་པའི་ལམ་གྱི་རིམ་པ་དག་ལ་ཡང་དག་པར་བསླབ་སྟེ་བློ་རྒྱུད་ལས་སུ་རུང༌། སེམས་རྒྱུད་ཆོས་དང་འདྲེས། བླ་མ་དམ་པ་དང་དགེ་བའི་བཤེས་གཉེན་གྱིས་གསུངས་པའི་གདམས་ངག་ལྟར་དུ་སོ་སོས་ངོ་ལྐོག་མེད་པར་ཉམས་ལེན་བྱས། ངོ་ལྐོག་བྱས་ན། ཕན་ཐོགས་མེད་པ་ཡིན། མིའི་མདུན་ལ་ཆོས་སྒྲུབ་ཀྱི་ཡོད་མདོག་གི་རྣམ་པ་དང༌། ཆོས་པ་ཡིན་མདོག་གི་རྣམ་པ་དང༌། སྒོམ་ཆེན་པ་ཡིན་མདོག་གི་རྣམ་པ་བསྟན་པས་མི་ཕན། དེས་ས་ལམ་ནི་བགྲོད་མི་ཐུབ། རྒྱལ་བ་སྲས་དང་བཅས་པ་ལ་ནི་བསླུ་མི་ཐུབ། རང་གི་ཚོགས་བསགས་སྒྲིབ་སྦྱོང་ལ་ནི་མི་འགྲོ་བས་ངོ་དང་ལྐོག་གཉིས་ལ་དབྱེ་བ་མེད་པ་ཞིག་དགོས་པ་ཡིན། ངོ་ལ་ཆོས་སྒྲུབ་ཀྱི་ཡོད་ན་ལྐོག་རང་ཉིད་གཅིག་པུར་གནས་པའི་ཚེ་ནའང་སེམས་ཆོས་དང་འདྲེས་འདུག་གམ་མི་འདུག །ལུས་ངག་ཡིད་གསུམ་དགེ་སྦྱོར་དང་འདྲེས་ཀྱི་འདུག་གམ་མི་འདུག །ཀུན་སློང་བཟང་པོ་འདུག་གམ་མི་འདུག །ཆོས་ནས་གསུངས་པ་ལྟར་དུ་ངས་ལག་ལེན་བྱེད་ཀྱི་འདུག་གམ་མི་འདུག་ཅེས་སོ་སོས་སོ་སོར་བརྟགས་ཤིང་དཔྱད་དེ་བཅོས་སྒྱུར་བྱས་པ་ཡིན་ན། དེ་ལ་ཆོས་ངོ་ལྐོག་མེད་པར་སྒྲུབ་པ་ཞེས་བརྗོད་པ་ཡིན། དེ་ལྟ་བུར་སྒྲུབ་ན་རང་ཉིད་ཚེ་འདི་ཕྱི་གཉིས་ཀ་ལ་སྐྱིད་པ་རེད། ཁྱེད་བཀའ་དྲིན་ཆེ་ཞེས་གཞན་ལ་བཀའ་དྲིན་ཆེ་བའི་ཡུས་སྒོམ་མི་དགོས་པར་རང་དྲིན་རང་ལ་ཆེ་བས་སོ་སོ་སོ་སོའི་བཀའ་དྲིན་རེད། རང་གིས་རང་ལ་སྐྱོབ་པ་རེད། བཅོམ་ལྡན་འདས་ཀྱིས་ཀྱང༌།
བདག་ཉིད་བདག་གི་མགོན་ཡང་ཡིན། །
ཞེས་གསུངས། བདག་ཉིད་བདག་གི་མགོན་རེད། སོ་སོ་སོ་སོས་མགོན་སྐྱོབ་བྱེད་དགོས་པ་ཡིན། རང་གི་ལུས་ངག་ཡིད་གསུམ་དེ་མི་དགེ་བའི་ལས་ལ་དུས་རྟག་ཏུ་སྤྱོད། བློ་རྒྱུད་ལས་སུ་མ་རུང་བ་ཞིག་གི་ནང་དུ་གནས། ད་འགག་ཅིག་ལ་ཐུག་པའི་ཚེ་ལྷ་བླ་མ་དཀོན་མཆོག་ལ་བདག་ལ་སྐྱོབ་རོགས་གནང༌། མགོན་རོགས་གནང༌། དཔུང་གཉེན་གནང་རོགས་གནང་ཟེར་དགོས་ན། སྐྱོབ་ཐུབ་པ་ཁག་པོ་རེད། ལྷ་བླ་མ་དཀོན་མཆོག་ལ་སྐྱོབ་པའི་ནུས་པ་ལྡན་ན་ཡང་རང་ངོས་ནས་སྐྱོབ་པའི་ཆ་རྐྱེན་མ་ཚང་བ་རེད། སྐྱོབ་པའི་ཆ་རྐྱེན་ཚང་བ་ལ་ཡོན་ཏན་དེ་དག་སྒྲུབ་དགོས་པ་ཡིན། དེ་ལྟ་མིན་པར་མགོན་རོགས་གནང༌། དཔུང་རོགས་གནང༌། སྐྱོབ་རོགས་གནང་ཟེར་ན་ཡང་དུས་ཚོད་ཅིག་ལ་ཐུག་དུས་མགོན་ཐུབ་ཀྱི་མེད། སྐྱབས་ཐུབ་ཀྱི་མ་རེད། སྐྱབས་ངོ་མ་དེ་གང་རེད་ཅེ་ན། ཆོས་རེད། ཆོས་ནས་བྱེད་དགོས་པ་འདི་དང་དེ་རེད། བྱེད་མ་རུང་བ་འདི་དང་དེ་རེད། བསམ་མནོ་འདི་འདྲ་ཞིག་གཏོང༌། སྤྱོད་པ་འདི་འདྲ་ཞིག་བྱེད། སྤང་རྒྱུ་འདི་རེད། བླང་རྒྱུ་འདི་རེད་ཅེས་སྤང་བླང་གི་རྣམ་གཞག་ཞིབ་ཅིང་ཕྲ་བར་གསུངས་པ་དེ་ཡིན། དེ་སེམས་ཐོག་ལ་ཚུར་སྦྱར་ན། སྐྱབས་ངོ་མ་རེད། དེ་མིན་སངས་རྒྱས་ཀྱི་སྐུ་ལ་ཕྱག་འཚལ། དམ་ཆོས་པོ་ཏི་མང་དག་ཅིག་མཆོད་གཤོམ་དུ་བརྩིགས། དེ་ལ་མཆོད་པ་ཕུལ། མཆོད་མེ་ཕུལ་བ་ཡིན་ན། བསོད་ནམས་ཀྱི་ཚོགས་ལྟ་ལོས་ཡོང་སྟེ་ཡིན་ན་ཡང་དངོས་གནས་སྐྱབས་དགོས་ས་ཞིག་ལ་སླེབ་དུས་སྐྱབས་ཐུབ་རྒྱུ་དཀའ་ལས་ཁག་པོ་རེད། དེའི་ནང་ནས་གསུངས་པ་དེ་སེམས་རྒྱུད་ལ་ཚུར་སྦྱར་དགོས་པ་ཡིན། དེ་ལ་ཆོས་སྒྲུབ་ཅེས་བརྗོད་པ་ཡིན། དེ་ལ་བརྟེན་ནས་ཚེ་ལ་ལོང་མེད་སྒྲུབ་ལ་བརྩོན་འགྲུས་བྱ་ཞེས་གསུངས་པའོ། །

ཞེས་བླ་མ་ནི་སངས་རྒྱས། བླ་མ་ནི་ཆོས། བླ་མ་ནི་དགེ་འདུན་དང༌། བླ་མ་ནི་ཡི་དམ། བླ་མ་ནི་ཆོས་སྐྱོང་རེད། དེའི་ཕྱིར་བླ་མ་མཆོད་པའི་ནང་ལ། ཁྱེད་ནི་བླ་མ། ཁྱེད་ནི་ཡི་དམ། ཁྱེད་ནི་མཁའ་འགྲོ་ཞེས་གསུངས། ཁྱེད་ནི་ཞེས་པ་དེ་ངོས་འཛིན་བྱེད་སའི་ཁྱད་གཞི་དེ་སུ་རེད་ཅེ་ན། བླ་མ་རེད། དེ་ལ་བརྟེན་ནས་སངས་རྒྱས་ཆོས་དང་དགེ་འདུན་གྱི་ཡོན་ཏན་ཡང་ང་ཚོ་ལ་འདིའི་སྟེང་ནས་ཐོབ་ཀྱི་ཡོད། ཡི་དམ་དང་ཆོས་སྐྱོང་གི་ཡོན་ཏན་ཡང་འདིའི་སྟེང་ནས་ཐོབ་ཀྱི་ཡོད། ས་ལམ་གྱི་ཡོན་ཏན་ཐམས་ཅད་ཀྱང་འདིའི་སྟེང་ནས་སྐྱེ་བས་བླ་མ་ཁྱེད་ཀུན་གྱི་བྱེད་པོ་རེད། སྔགས་ཀྱི་སྐབས་སུ་ནི་རྡོ་རྗེ་སློབ་དཔོན་དེ་དཀྱིལ་འཁོར་ཐམས་ཅད་ཀྱི་སྤྲོ་བསྡུའི་བྱེད་པོ་བླ་མ་ཁྱེད་ཡིན་སྙམ་ནས། ཡོན་ཏན་སངས་རྒྱས་དང་མཉམ་པ་ཡིན་ན་ཡང་སྐུ་དྲིན་སངས་རྒྱས་ལས་ཀྱང་ལྷག་པ་ཡིན་ཟེར་བའི་རྒྱུ་མཚན་ཡང་འདི་ཡི་སྟེང་ནས་འགྲོ་བ་ཡིན། དེ་ཡང་དགོས་པ་དང་ཕན་ཡོན་ཡོན་ཏན་གྱི་ཆ་ནས་བསམ་མནོ་བཏང་བ་མ་གཏོགས་གཞན་བླ་མ་སངས་རྒྱས་ལས་ལྷག་པ་ཡིན་པའི་རྒྱུ་མཚན་མེད། རྒྱུ་མཚན་ཡང་བླ་མའི་ཐུགས་རྗེ་ལ་བརྟེན་ནས་རང་རྒྱུད་ལ་ཉོན་མོངས་དང་རྣམ་རྟོག་ངན་པ་ཐམས་ཅད་ཞི། རང་རྒྱུད་ལ་ཡོད་པའི་ཡོན་ཏན་ཐམས་ཅད་གོང་དུ་འཕེལ་ཐུབ་པའི་རྐྱེན་དེ་བླ་མ་ལ་བརྟེན་ནས་བྱུང་བའི་རྒྱུ་མཚན་གྱིས་ཡིན། ད་ལྟ་ང་ཚོས་སངས་རྒྱས་མཇལ་དགོས་ན། སངས་རྒྱས་ལོངས་སྤྱོད་རྫོགས་པའི་སྐུ། སངས་རྒྱས་སྤྲུལ་པའི་སྐུ། སངས་རྒྱས་ཆོས་ཀྱི་སྐུ་རྣམས་མཇལ་མི་ཐུབ་པ་རེད། དེ་དག་མཇལ་བར་བྱེད་པ་ལ་ང་ཚོས་དང་པོར་བླ་མ་བསྟེན། བླ་མའི་མདུན་ནས་གདམས་ངག་ཞུས་པས་བླ་མའི་ཐུགས་རྗེ་ལ་བརྟེན་ཏེ་དེ་དག་གི་ཡོན་ཏན་ང་རང་ཚོ་ལ་འཐོབ་ཐུབ་པ་རེད། དེའི་ཕྱིར་ན་བླ་མ་ཁྱེད་ཡོན་ཏན་སངས་རྒྱས་དང་མཉམ་པ་ཡིན་ན་ཡང་སྐུ་དྲིན་གྱི་ཐོག་ལ་བསམ་ན། ཁྱེད་སངས་རྒྱས་ལས་ཀྱང་ལྷག་པ་རེད་ཅེས་ཡོན་ཏན་དེ་རང་གི་སེམས་རྒྱུད་ཀྱི་ཐོག་ནས་ཤེས་རབ་ཀྱིས་དྲངས་པའི་ཤེས་ནས་དད་པ་ཐོབ་པའི་ངེས་ཤེས་ཤིག་དགོས་ཀྱི་འདུག །དེའི་ཕྱིར་བླ་མ་ཞེས་པ་དེ་དང་པོ་རྩ་བ་ཆེན་པོ་ཡིན་གསུངས་པ་ནི་འདིའོ། །

དཀོན་མཆོག་གསུམ་ལ་མ་བརྟེན་པའི། །
ཆོས་པ་བཟང་ཡང་མུ་སྟེགས་ཡིན། །
ཞེས་དཀོན་མཆོག་གསུམ་ལ་མ་བརྟེན་པར་ཕར་བལྟས་ན་ཆོས་པ་བཟང་པོར་སྣང་ནའང་དོན་ལ་མུ་སྟེགས་པ་ཡིན་གསུངས་པའོ། །དེའི་ཕྱིར་སྐྱབས་སུ་འགྲོ་དགོས་པ་ཡིན། སྐྱབས་སུ་འགྲོ་བ་ལ་ཡང་རིམ་པ་མང་པོའི་ཐོག་ནས་བསམ་མནོ་བཏང་དགོས་པ་ཡིན། ཐེག་པ་ཆེན་པོའི་སྐབས་ཐེག་ཆེན་གྱི་སྐྱབས་འགྲོ་བཤད་པ་ན་མར་གྱུར་སེམས་ཅན་དེ་ཐམས་ཅད་སྙིང་རེ་རྗེ་ཞེས་བཤད་ཡོང་དུས། རང་གི་སྡུག་བསྔལ་ལས་དཔེ་བླངས་ཏེ་གཞན་གྱི་སྡུག་བསྔལ་ལ་བཟོད་དུ་མེད་པའི་བསམ་མནོ་གཏང་སྟེ་མི་བཟོད་པའི་ཚོར་བ་ཞིག་སྐྱེ་དགོས་ལ། རང་གི་བདེ་སྐྱིད་ལ་དཔེ་བླངས་ཏེ་དེ་ལ་དགའ་བ་དང་སྤྲོ་བ་ཇི་ལྟར་ཡོད་པ་དེ་ལྟར་སེམས་ཅན་གྲངས་མེད་ཚད་མེད་ཀྱི་རྒྱུད་ལ་ཡོད་པའི་བདེ་སྐྱིད་ལ་ཡང་དགའ་བ་དང་སྤྲོ་བ་སྐྱེ་དགོས་པ་ཡིན། དེ་སྐྱེས་ན་གཞན་ལ་ཕྲག་དོག་དང་ཞེ་སྡང་དང་ཁོང་ཁྲོ་དང་ང་རྒྱལ་དང་སེར་སྣ་སོགས་མ་རུངས་པའི་བློ་སྐྱེ་བའི་གོ་སྐབས་མེད་པ་ཡིན། དེ་དག་མི་སྐྱེ་བའི་ཐབས་ལ་གང་དགོས་ཤེ་ན། རང་བདེ་བ་འདོད་པ་བཞིན་གཞན་སེམས་ཅན་ཐམས་ཅད་ཀྱང་བདེ་བ་འདོད། རང་སྡུག་བསྔལ་མི་འདོད་པ་བཞིན་སེམས་ཅན་གཞན་ཐམས་ཅད་ཀྱང་སྡུག་བསྔལ་མི་འདོད་པ་སྒོ་ཀུན་ནས་མཚུངས་པས་རང་གཅིག་པུའི་བདེ་བ་སྒྲུབ་སྟེ་གཞན་གྱི་བདེ་བ་མཐོང་དུས། དེ་ལ་ཕྲག་དོག་སོགས་བྱེད་མི་རུང༌། རང་ལ་སྡུག་བསྔལ་འབྱུང་དུས་དེ་དོར་བའི་ཐབས་ལ་འབད་ལ་གཞན་གྱི་སྡུག་བསྔལ་ཡལ་བར་དོར་ཏེ་ཇི་མི་སྙམ་པར་གནས་ན་དེ་ཡང་མི་རུང་བའོ། །དཔེར་ན་རང་ལ་ལུས་ལ་ན་ཚ་ཐེབས་པ་ན་དཀའ་ངལ་གང་འདྲ་འདུག །སེམས་ལ་དཀའ་ངལ་འཕྲད་དུས་མི་བཟོད་པའི་སྡུག་བསྔལ་ཅི་འདྲ་འདུག །དེ་དང་མཚུངས་པ་སྡུག་བསྔལ་དེ་ལྡན་གྱི་སེམས་ཅན་གྲངས་མང་བ་ཚད་མེད་པ་དེ་དག་ཐམས་ཅད་ཕ་མར་མ་གྱུར་པ་གཅིག་ཀྱང་མེད། ཕ་མར་གྱུར་པའི་རིང་ཚེ་འདིའི་ཕ་མ་དང་འདྲ་བའི་དྲིན་ཆེན་ཤ་སྟག་རེད། རང་དང་དེ་ཐམས་ཅད་ཀྱི་དམིགས་ཡུལ་བདེ་བ་འདོད་པ་དང་སྡུག་བསྔལ་མི་འདོད་པ་ནི་སྒོ་ཀུན་ནས་མཚུངས་པར་འདུག་པས། ང་ད་རེས་དམ་པའི་ཆོས་ཐོས། བླ་མ་བཟང་པོ་མཇལ། སངས་རྒྱས་ཀྱི་བསྟན་པ་མཇལ་ནས་དགེ་བའི་ཕན་ཡོན་ཤེས། སྡིག་པའི་ཉེས་དམིགས་ཤེས། བྱང་ཆུབ་སེམས་སྐྱེ་བའི་ཕན་ཡོན་ཤེས། མ་སྐྱེས་པའི་ཉེས་དམིགས་ཤེས་པའི་གང་ཟག་ཅིག་གིས་དེ་དག་ཡལ་བར་དོར་ན། གཏན་ནས་འགྲིག་གི་མི་འདུག་བསམ་པའི་གློ་སྙིང་བྲང་གསུམ་སེམས་ཅན་ཐམས་ཅད་ལ་སྤྲད་དེ་སེམས་ཅན་ཐམས་ཅད་ལ་བྱམས་སྙིང་རྗེ་བྱང་ཆུབ་ཀྱི་སེམས་ཀྱི་ཐོག་ནས་ཕན་བདེ་སྒྲུབ་དགོས་པ་ཡིན། གཞན་གྱི་དོན་ཞེས་པའང་དེ་ཡིན། དེ་ལྟར་སྒྲུབ་ན་སྐྱབས་ཡུལ་དམ་པ་རྣམས་ལ་མཆོད་པ་མཆོག་ཏུ་གྱུར་པའང་དེ་ཡིན་ཏེ། སེམས་ཅན་ཐམས་ཅད་ལ་བདེ་སྐྱིད་བྱུང་ན་སངས་རྒྱས་ཐམས་ཅད་ཐུགས་མཉེས་ལ་སྡུག་བསྔལ་བྱུང་ན་ཐུགས་མ་མཉེས་པའི་ཕྱིར་མཆོད་པ་མཆོག་ཏུ་གྱུར་པ་ཡིན། རང་གིས་བར་དུ་བསོད་ནམས་ཀྱི་ཚོགས་བསགས་པ་དང་སྒྲིབ་པ་སྦྱང་བའང་དེ་ལས་ལྷག་པ་མེད། མཐར་ཐུག་རྣམ་མཁྱེན་གྱི་གོ་འཕང་ཐོབ་པ་ལ་ཡང་བྱང་ཆུབ་ཀྱི་སེམས་ལ་མ་བརྟེན་པར་རྣམ་མཁྱེན་གྱི་གོ་འཕང་ཐོབ་ཏུ་མེད་པས་ན་འགོ་མཇུག་བར་གསུམ་གང་ནས་བསམ་མནོ་བཏང་བ་ཡིན་ན་ཡང་བྱང་ཆུབ་སེམས་ལས་ལྷག་པ་མེད། དེའི་ཕྱིར་ཐེག་པ་ཆེན་པོའི་སྐབས་དང་པོ་བྱང་ཆུབ་ཏུ་སེམས་བསྐྱེད། དེ་ནས་སྐྱབས་སུ་འགྲོ་བ་སྟེ། སེམས་ཅན་ཐམས་ཅད་ཀྱི་དོན་དུ་བདག་གིས་སངས་རྒྱས་ཀྱི་གོ་འཕང་འཐོབ་པར་བྱ། དེའི་ཆེད་དུ་སངས་རྒྱས་ཆོས་དང་དགེ་འདུན་གསུམ་ལ་སྐྱབས་སུ་འགྲོའོ་སྙམ་པའི་དམིགས་ཡུལ་ཆེད་དུ་བྱ་བ་སེམས་ཅན་ཐམས་ཅད་ཀྱི་དོན། དེ་དག་བདེ་བ་འཐོབ་པ་དང་སྡུག་བསྔལ་འབྲལ་བའི་ཐབས་ལ་བསམ་སྟེ་དེའི་ཆེད་དུ་སངས་རྒྱས་ཆོས་དང་དགེ་འདུན་གསུམ་ལ་སྐྱབས་སུ་འགྲོ་བ་ནི་ཐེག་ཆེན་གྱི་སྐྱབས་འགྲོ་ཡིན། དེའི་ཕྱིར་སངས་རྒྱས་ཀྱི་གོ་འཕང་ལ་འགོད་པ་ལ་སྐྱབས་ཡུལ་རྣམས་ལ་ལམ་སྣ་འདྲེན་ཞིང་བར་ཆད་སེལ་བའི་དོན་དུ་སྐྱབས་སུ་འགྲོའོ་སྙམ་དུ་བསམས་དགོས། རྗེ་བཙུན་མི་ལ་རས་པས།
རྗེ་བྱིན་རླབས་ཅན་ལ་གསོལ་བ་འདེབས། །
སྐྱབས་དཀོན་མཆོག་གསུམ་དེ་རྒྱབ་རྟེན་བཟང༌། །
ཞེས་དང་པོ་བླ་མ་ལ་སྐྱབས་སུ་འགྲོ་དགོས་པར་གསུངས། རྗེ་བྱིན་རླབས་ཅན་ལ་གསོལ་བ་འདེབས་ཞེས་བྱིན་རླབས་ཅན་ཞེས་ཅན་སྒྲ་དེ་སུམ་རྟགས་ཀྱི་སྐབས་ཅན་ལྡན་བདག་པོའི་སྒྲ་རེད། བྱིན་རླབས་རྒྱུད་ལྡན་གྱི་གང་ཟག་ཅིག་དགོས་ཀྱི་ཡོད། དེ་སུ་རེད་ཅེ་ན། རང་ཉིད་ལྟ་སྒོམ་སྤྱོད་གསུམ་གྱི་ཆོས་རྣམ་པར་དག་པ་སྔོན་དུ་སོང༌། དེའི་སྟོབས་ཀྱིས་གཞན་ལ་ཕན་པར་འདོད་པའི་བྱ་བ་རྩོམ་མཁན་ཞིག་ཡོད་ན། དེ་བྱིན་རླབས་རྒྱུད་ལྡན་གྱི་གང་ཟག་རེད། དེ་སུ་རེད་ཅེ་ན། བླ་མ་རེད། དེ་སུ་རེད་ཅེ་ན། གདམས་ངག་སྤྲོད་མཁན་གྱི་གང་ཟག་རེད། དཔེར་ན། རྗེ་བཙུན་མི་ལ་རས་པ་ལྟ་བུ། དྭགས་པོ་རིན་པོ་ཆེ་ལྟ་བུ། རང་གི་རྩ་བའི་བླ་མ་ལ་བྱིན་རླབས་ཅན་ཞེས་གསུངས་ཀྱི་ཡོད། ང་ཚོ་བྱིན་རླབས་ལ་རེ་བ་ཡོད། བྱིན་རླབས་ལ་རེ་བས་རྗེ་བྱིན་རླབས་ཅན་ལ་གསོལ་བ་འདེབས་ཞེས་གསོལ་བ་བཏབ་པ་ཡིན། བླ་མ་དང་ཇོ་བོ་དཀོན་མཆོག་གསུམ་ཞེས་པའི་བླ་མ་དེ་རྗེ་བྱིན་རླབས་ཅན་ལ་གསོལ་བ་འདེབས་ཞེས་གསུངས་པ་དེ་ཡིན་ལ། ཇོ་བོ་དཀོན་མཆོག་གསུམ་ཞེས་པ་ནི་སྐྱབས་དཀོན་མཆོག་གསུམ་དེ་རྒྱབ་རྟེན་བཟང་ཞེས་པ་དེ་ཡིན། དེ་ཡང་དཔེར་ན། ས་གནས་ཉེན་ཚབས་ཆེ་བ་ལ་འགྲོ་དགོས་ན། དཔུང་དམག་འཁོར་དང་མཉམ་དུ་འགྲོ་ན་སོ་སོའི་སྲོག་ལ་ཉེན་ཁ་མེད་པ་བཞིན། འདི་ནས་བྱང་ཆུབ་སྒྲུབ་པ་ལ་སྐྱབས་དཀོན་མཆོག་གསུམ་ལས་ལྷག་པའི་རྒྱབ་རྟེན་མེད། དེའི་ཕྱིར་ན་སྐྱབས་དཀོན་མཆོག་གསུམ་དེ་རྒྱབ་རྟེན་བཟང་ཞེས་གསུངས་པའོ། །ཡི་དམ་ལྷ་ཚོགས་དཔའ་བོ་མཁའ་འགྲོ་ཆོས་སྐྱོང་སྲུང་མ་ཞེས་གསུངས་པ་ནི། དེ་ཡང་མི་ལ་རས་པས།
གནས་གསུམ་གྱི་མཁའ་འགྲོ་སྐྱེལ་མ་འོང༌། །
ཞེས་གནས་གསུམ་གྱི་དཔའ་བོ་མཁའ་འགྲོ་ཆོས་སྐྱོང་སྲུང་མ་རྣམས་ང་ཚོ་ལ་སྐྱེལ་མར་ཡོང་གི་ཡོད། ས་གནས་ཉེན་ཁ་ཅན་ལ་འགྲོ་དགོས་ན། སྐྱེལ་མ་དཔའ་བོ་མེ་མདའ་དང་མཚོན་ཆ་སོགས་ཐོགས་པའི་གང་ཟག་དང་ལྷན་དུ་འགྲོ་བ་ན་རང་ལ་སྲུང་སྐྱོབ་ཡག་པོ་ཡོང་བ་བཞིན། ཆོས་བཞིན་སྒྲུབ་ན་ཡི་དམ་དཔའ་བོ་མཁའ་འགྲོ་ཆོས་སྐྱོང་ལྷ་ཚོགས་ཐམས་ཅད་ཀྱིས་ཀྱང་ང་རང་ཚོ་ལ་སྐྱེལ་མ་མཛད། མགོན་སྐྱོབ་དང་དཔུང་གཉེན་མཛད་ཀྱི་རེད་ཅེས་གསུངས་པའོ། །མར་གྱུར་སེམས་ཅན་ཐམས་ཅད་འཁོར་བའི་རྒྱ་མཚོར་སྡུག་བསྔལ་བར་གྱུར་པ་སྙིང་རེ་རྗེ་ཞེས་གསུངས་པ་ལ། དེ་ཡང་མི་ལ་རས་པས།
གྲོགས་བྱང་ཆུབ་སེམས་དང་ལྷན་ཅིག་འགྲོགས། །
ཞེས་གསུངས། གྲོགས་བཟང་ཤོས་དེ་གང་རེད་ཅེ་ན། བྱང་ཆུབ་ཀྱི་སེམས་རེད། བྱང་ཆུབ་ཀྱི་སེམས་བསྐྱེད་པ་ལ་སེམས་ཅན་ཐམས་ཅད་མར་ཤེས་པ། དྲིན་དྲན་པ། དྲིན་གཟོ་བ། ཡིད་དུ་འོང་བའི་བྱམས་པ། སྙིང་རྗེ། དེ་ནས་ལྷག་བསམ་རྣམ་དག་ཅེས་རྒྱུ་དྲུག་སྔོན་དུ་སོང༌། དེའི་སྟོབས་ཀྱིས་འབྲས་བུ་བྱང་ཆུབ་ཀྱི་སེམས་བསྐྱེད་པ་སྟེ་ཇོ་བོ་བཀའ་གདམས་པའི་མན་ངག་ལུགས་ལ་རྒྱུ་འབྲས་མན་ངག་བདུན་གྱི་སྒོ་ནས་བྱང་ཆུབ་ཀྱི་སེམས་རྒྱུད་ལ་བསྐྱེད་པ། དེའི་བསླབ་པ་ལ་སློབ་པ་ཞེས་གསུངས་པའི་རིམ་པ་དེ་འདིའི་སྟེང་ནས་ཡོང་གི་ཡོད། གྲོགས་བྱང་ཆུབ་སེམས་དང་ལྷན་ཅིག་འགྲོགས་གསུངས་པ་དེ་ཡིན། བྱང་ཆུབ་ཀྱི་སེམས་དང་ལྷན་ཅིག་ཏུ་གྲོགས་ན་ང་ཚོ་ལ་བར་ཆད་མི་ཡོང་བ་ཡིན། ཆོས་སྒྲུབ་པའི་སྐབས་འདྲེ་ཡི་གནོད་པ་འབྱུང་པོ་སོགས་ཀྱི་གནོད་པ་མང་པོ་ཡོང་གི་འདུག ཅེས་བཤད་སྲོལ་ཡོད། མི་ཡོད་སར་འདྲེ་ཡོད་ཆོག་པ་རེད། ཚང་མ་མཉམ་དུ་བསྡད་ཡོད་པ་རེད། ཡིན་ན་ཡང་འདྲེ་དང་ལྷ་སྲིན་འབྱུང་པོ་སོགས་ཀྱིས་གནོད་པ་སྐྱེལ་དང་མ་བསྐྱལ་དེ་སོ་སོ་ལ་རག་ལས་པ་ཡིན་ཏེ། རང་སེམས་ལས་སུ་མ་རུང་བར་བྱས། ཆགས་སྡང་ཉོན་མོངས་པའི་རྣམ་རྟོག་ཁོ་ནའི་དབང་དུ་སོང་བ་ཡིན་ན་གནོད་པ་ལོས་ཡོང༌། ཡོང་གི་རེད། བླ་མ་གོང་མ་ཐམས་ཅད་ཐུགས་དམ་ཉམས་རྟོགས་གི་རྩལ་སྦྱོང༌། རིན་ཆེན་བྱང་ཆུབ་ཀྱི་སེམས་རྒྱུད་ལ་བསྐྱེད། བདག་གཞན་བརྗེ་བའི་བྱང་ཆུབ་ཀྱི་སེམས་ཁོ་ན་ལ་བློ་སྦྱངས་ཏེ་རང་དོན་ཡལ་བར་དོར། གཞན་དོན་ཁོ་ན་དོན་དུ་གཉེར་ནས་བྱང་ཆུབ་ཀྱི་སེམས་སྐྱེས་པའི་སེམས་དཔའ་རྣམས་ལ་ལྷ་སྲིན་སྡེ་བརྒྱད་ཐམས་ཅད་བྲན་དུ་བཀོལ། ཅི་གསུངས་བཀའ་བཞིན་སྒྲུབ་པར་བྱས། རྗེ་བཙུན་མི་ལ་རས་པའི་རྣམ་ཐར་དང་རྒྱལ་བ་རྒོད་ཚང་པའི་རྣམ་ཐར་དང་ཁ་རག་སྒོམ་ཆུང་སོགས་བླ་མ་གོང་མ་རྣམས་ཀྱི་རྣམ་ཐར་ལ་གཟིགས་ན་མཁྱེན་གྱི་རེད། ལྷ་སྲིན་སྡེ་བརྒྱད་བྲན་དུ་བཀོལ། ཅི་གསུང་བཀའ་སྒྲུབ། ཆོས་སྐྱོང་བའི་སྲུང་མ་ལ་ཡར་བསྐོས་པ་སོགས་ལྷ་སྲིན་སྡེ་བརྒྱད་ཐམས་ཅད་རང་དབང་དུ་བསྡུས་ཐུབ་པ་དེ་ཡང་བྱང་ཆུབ་ཀྱི་སེམས་ལ་བརྟེན་ནས་བྱུང་བ་རེད། དེར་བརྟེན་རྗེ་བཙུན་མི་ལ་རས་པས།

དེའི་སྐུ་པང་དུ་ཡུམ་རྡོ་རྗེ་སྙེམས་མ་དཀར་མོ་གྲི་གུག་དང་ཐོད་པ་འཛིན་པས་ཡབ་ལ་འཁྱུད་ཅིང་པདྨའི་འདུག་སྟངས་ཀྱིས་བཞུགས་པ། གཉིས་ཀའང་མཚན་དཔེས་སྤྲས་ཤིང་རུས་པ་དང་རིན་པོ་ཆེའི་རྒྱན་གྱིས་བརྒྱན་པ། ཞི་ཞིང་སྒེག་པའི་ཉམས་དང་ལྡན་ཞིང་དར་གྱི་ན་བཟའ་གསོལ་བ། སྣང་ལ་རང་བཞིན་མེད་པའི་སྤྱི་བོར་ཨོཾ། མགྲིན་པར་ཨཱཿ ཐུགས་ཀར་ཧཱུྃ་གིས་མཚན་པའི་ཧཱུྃ་ལས་འོད་ཟེར་འཕྲོས་པས་རང་བཞིན་གྱི་གནས་ནས་སྒོམ་དང་འདྲ་བའི་ཡེ་ཤེས་པ་སྤྱན་དྲངས། བཛྲ་ས་མཱ་ཛ༔ ཛཿཧཱུྃ་བཾ་ཧོ༔ དམ་ཚིག་པ་དང་དབྱེར་མེད་དུ་གྱུར་ཅེས་མོས་ཏེ། བཅོམ་ལྡན་འདས་བདག་དང་སེམས་ཅན་ཐམས་ཅད་ཀྱིས་འཁོར་བ་ཐོག་མ་མེད་པ་ནས་བསགས་པའི་སྡིག་སྒྲིབ་ཉེས་ལྟུང་ཐམས་ཅད་བྱང་ཞིང་དག་པར་མཛད་དུ་གསོལ། ཞེས་གསོལ་བཏབ་པས་རྡོ་རྗེ་སེམས་དཔའི་ཐུགས་ཀར་ཟླ་བའི་སྟེང་དུ་ཧཱུྃ་དཀར་པོའི་མཐར་ཡི་གེ་བརྒྱ་པའི་སྔགས་ཕྲེང་དཀར་པོས་འཁོར་བ་ལས་འོད་ཟེར་འཕྲོས། ཕྱོགས་བཅུའི་དེ་བཞིན་གཤེགས་པ་ཐམས་ཅད་ཀྱི་ཐུགས་ཀ་ནས་ཡེ་ཤེས་ཀྱི་བདུད་རྩི་བཀུག །རྡོ་རྗེ་སེམས་དཔའ་ཡབ་ཡུམ་གྱི་སྤྱི་བོ་ནས་ཞུགས། ཐུགས་ཀར་ཐིམ། སླར་ཡང་ཐུགས་ཀ་ནས་ཡེ་ཤེས་ཀྱི་བདུད་རྩིའི་རྒྱུན་བབས། སྦྱོར་མཚམས་དང་ཞབས་ཀྱི་སོར་མོ་ནས་བྱུང་བ་རང་གི་ཚངས་བུག་ནས་ཞུགས་ཏེ་ལུས་ཐམས་ཅད་གང༌། ནད་གདོན་སྡིག་སྒྲིབ་ཐམས་ཅད་དབང་པོའི་སྒོ་དང་བ་སྤུའི་བུ་ག་ནས་ཕྱིར་ཟགས་པས་ལུས་ཤེལ་གོང་གཡའ་དག་པ་ལྟ་བུ་གསལ་ཞིང་དྲི་མ་མེད་པར་གྱུར། ཞེས་མོས་ཤིང༌།
ཞེས་སྒོམ་རིམ་དེ་དག་གོ་སླ། གང་ལྟར་སྣང་ལ་རང་བཞིན་མེད་པའི་རྡོ་རྗེ་སེམས་དཔའ་ཞིག་སྒོམ་དགོས། སྤྱིར་ལྷ་གང་དང་གང་སྒོམ་ནའང་ལྷ་རྫས་སུ་གྲུབ་པའམ་བདེན་པར་གྲུབ་པ། ལྷ་རང་བཞིན་གྱིས་གྲུབ་པའམ་མཚན་མར་གྲུབ་པར་སྒོམ་ན། ལྷའི་བྱིན་རླབས་མི་འཇུག་པའོ། །འོ་ན་ཇི་ལྟ་བུར་སྒོམ་དགོས་སྙམ་ན། སྒྱུ་མ་ལྟ་བུ། རྨི་ལམ་ལྟ་བུ། དྲི་ཟའི་གྲོང་ཁྱེར་ལྟ་བུ། སྣང་ལ་རང་བཞིན་གྱིས་མ་གྲུབ་པ། སྣང་ཙམ། གསལ་ཙམ། རིག་ཙམ་ཞིག་གི་བདག་ཉིད་དུ་སྒོམ་དགོས་ཞེས་ཀུན་མཁྱེན་མི་ཕམ་རིན་པོ་ཆེས་བསྐྱེད་རིམ་རྣམ་བཤད་ཟབ་མོའི་ནང་གསུངས།

ཅེས་བླ་མ་སྐུ་བཞིའི་གངས་རི་ལྟ་བུ་དེ་ལ་ཉི་མ་ཤར་བ་ན་གངས་རི་བཞུས་ཏེ་དེ་ལས་ཆུ་བྱུང་བ་བཞིན། མོས་གུས་ཀྱི་ཉི་མ་མ་ཤར་ན་བླ་མའི་སྐུ་གསུང་ཐུགས་ཀྱི་བྱིན་རླབས་རང་རྒྱུད་ལ་མི་འཇུག་པ་ཡིན། བླ་མའི་བྱིན་རླབས་རང་གི་རྒྱུད་ལ་འཇུག་མི་འཇུག་དེ་རང་ལ་མོས་གུས་རྣམ་པར་དག་པ་དང་མོས་གུས་ཟབ་པོ་ཞིག་ཡོད་མེད་ལ་རག་ལུས་པས་མོས་གུས་རྩལ་དུ་འདོན་དགོས་གསུངས། བླ་མ་ལ་མོས་གུས་བསྟེན། བླ་མའི་བྱིན་རླབས་རང་རྒྱུད་ལ་འཇུག་ཡོང་དུས་བདེ་སྟོང་གཉིས་སུ་མེད་པའི་ཕྱག་རྒྱ་ཆེན་པོ་འོད་གསལ་ལྷན་ཅིག་སྐྱེས་པའི་ཡེ་ཤེས་དེ་དེའི་ཚེ་ངོ་འཕྲོད་ཀྱི་ཡོད། དེ་འཕྲོད་མ་འཕྲོད་ཀྱང་བླ་མ་ལ་རག་ལས་པས་ན་ལམ་ཟབ་མོ་བླ་མའི་རྣལ་འབྱོར་ཞེས་གསུངས། ལམ་ཟབ་མོ་གསུངས་དགོས་པའང་བླ་མ་ལ་བསྟེན། གདམས་ངག་ཞུ། དེ་ཉམས་སུ་ལེན། དེ་ལ་གནད་དུ་བསྣུན་ན་སྟན་ཐོག་འདི་ཉིད་ལ་གྲོལ་ཐུབ་པ། སྡོད་ས་འདི་ཉིད་དུ་གྲོལ་ཐུབ་པ། ཐུན་འདི་ཉིད་ལ་གྲོལ་ཐུབ་པའི་ནུས་པའང་འདི་ཡི་སྟེང་ནས་ཡོང་བའི་ཕྱིར་ན་ལམ་ཟབ་མོ་བླ་མའི་རྣལ་འབྱོར་ཞེས་གསུངས་པའོ། །
ཨོཾ་སྭ་བྷཱ་ཝ་ཤུདྡྷ༔སརྦ་དྷརྨཿས་བྷཱ་ཝ་ཤུདྡྷོ྅ཧཾ། ཆོས་ཐམས་ཅད་རང་བཞིན་གྱིས་འོད་གསལ་བ་སྟོང་པ་ཉིད་གྱུར། སྟོང་པའི་ངང་ལས་རྡོ་རྗེའི་ས་གཞི་ར་བ་དྲྭ་བ་གུར་བླ་རེ་མེ་རི་དང་བཅས་པའི་སྲུང་འཁོར་གྱི་ནང་དུ་སྣ་ཚོགས་པདྨ་འདབ་མ་བརྒྱད་ལྟེ་བར་ཉི་མའི་དཀྱིལ་འཁོར། དེའི་དབུས་སུ་ཧཱུྃ་ལས་རྡོ་རྗེ་ཧཱུྃ་གིས་མཚན་པ་ལས་འོད་ཟེར་འཕྲོས། འགྲོ་དོན་བྱས། སླར་འདུས། ཡོངས་སུ་གྱུར་པ་ལ་རང་ཉིད་བཅོམ་ལྡན་འདས་འཁོར་ལོ་སྡོམ་པ་སྐུ་མདོག་སྔོན་པོ་ཞལ་གཅིག་སྤྱན་གསུམ་པ་མཆེ་བ་ཅུང་ཟད་གཙིགས་པ། སྐྲ་མཚམས་སུ་རྡོ་རྗེའི་ཕྲེང་བས་སྦྲེལ་བའི་ཐོད་སྐམ་རིན་པོ་ཆེའི་རྩེ་ཕྲན་ཅན་ལྔས་བརྒྱན་ཅིང་རལ་པའི་ཅོད་པན་གྱི་རྩེ་མོར་ནོར་བུ་མདུན་དུ་སྣ་ཚོགས་རྡོ་རྗེ་གཡོན་དུ་ཟླ་བ་ཕྱེད་པས་སྤྲས་པ། ཕྱག་གཉིས་ཀྱིས་རྡོ་རྗེ་དང་དྲིལ་བུ་འཛིན་པས་ཡུམ་ལ་འཁྱུད་ལ། ཞབས་གཡོན་བསྐུམས་པས་འཇིགས་བྱེད་ཀྱི་མགོ་བོ་བལྟ་བ་སྟེ་བརྫིས་ཤིང་གཡས་བརྐྱངས་པས་དུས་མཚན་གྱི་ནུ་བར་དུ་མནན་པ། སྟག་ལྤགས་ཀྱི་ཤམ་ཐབས་གྲོལ་ཞིང་ཐོད་རློན་ལྔ་བཅུའི་དོ་ཤལ་འཕྱང་བ། ཕྱག་རྒྱ་དྲུག་གིས་བརྒྱན་ཅིང་གར་གྱི་ཉམས་དགུ་དང་ལྡན་པ། དེའི་པང་དུ་ཡུམ་རྡོ་རྗེ་རྣལ་འབྱོར་མ་སྐུ་མདོག་དམར་མོ་ཞལ་གཅིག་སྤྱན་གསུམ་པ། དབུ་སྐྲ་སིལ་བུ་གྲོལ་བའི་མཚམས་སུ་ཐོད་པ་སྐམ་པོ་ལྔས་སྤྲས་པ། ཕྱག་གཉིས་ཀྱིས་གཡོན་ཐོད་པ་བདུད་རྩིས་གང་བ་བསྣམས་ནས་ཡབ་ཀྱི་མགུལ་ལ་འཁྱུད་དེ་སྟོབ་ཅིང་གཡས་སྡིགས་མཛུབ་དང་བཅས་པས་རྡོ་རྗེ་གྲི་གུག་འཕྱར་བ། ཞབས་གཡོན་བརྐྱང་ཞིང་གཡས་བསྐུམ་པས་ཡབ་ལ་འཁྲིལ་བ། བདེ་བ་ཆེན་པོའི་ཉམས་ཀྱིས་སྒེག་ཅིང་ལང་ཚོས་མཛེས་པ། གཅེར་མོ་ཐོད་སྐམ་ལྔ་བཅུའི་དོ་ཤལ་དང་རུས་པའི་ཕྱག་རྒྱ་ལྔས་བརྒྱན་པ། གཉིས་ཀའང་མཚན་དཔེས་སྤྲས་ཤིང་བསྐལ་པའི་མེ་ལྟ་བུའི་གཟི་བརྗིད་ཅན། ཐབས་ཤེས་གཉིས་སུ་མེད་པར་སྙོམས་པར་འཇུག་པ། དེའི་དཔྲལ་བར་ཟླ་བ་ལ་ཨོཾ་དཀར་པོ། མགྲིན་པར་པདྨ་ལ་ཨཱཿདམར་པོ། སྙིང་གར་ཉི་མ་ལ་ཧཱུྃ་སྔོན་པོས་མཚན་པར་གྱུར། ཕེཾ་ཞེས་པའི་སྒྲ་དང་སྙིང་གའི་ཧཱུྃ་གི་འོད་ཀྱིས་ཕྱོགས་བཅུའི་དཔའ་བོ་དང་རྣལ་འབྱོར་མ་ཐམས་ཅད་སྤྱན་དྲངས བཛྲ་ས་མཱ་ཛ། ཛ་ཧཱུྃ་བྃ་ཧོཿ དམ་ཚིག་པ་དང་དབྱེར་མེད་དུ་གྱུར། ཞེས་རང་བདེ་མཆོག་ལྷན་སྐྱེས་སུ་བསྒོམ་པའམ། གཞན་ཡང་ཡི་དམ་གྱི་ལྷ་གང་ཡིན་རེ་བསྒོམས་ལ། རང་གི་མདུན་གྱི་ནམ་མཁར་སེང་གེས་བཏེག་པའི་རིན་པོ་ཆེའི་ཁྲིའི་སྟེང་དུ་སྣ་ཚོགས་པདྨ་དང་ཉི་ཟླའི་གདན་ལ། ངོ་བོ་རྩ་བའི་བླ་མ་ཡིན་པ་ལ་བཅོམ་ལྡན་འདས་རྡོ་རྗེ་འཆང་སྐུ་མདོག་སྔོ་སངས་ཞལ་གཅིག་ཕྱག་གཉིས་རྡོ་རྗེ་དང་དྲིལ་བུ་བསྣོལ་ཐབས་སུ་འཛིན་པས་ཡུམ་ལ་འཁྱུད་ཅིང་ཞབས་སྐྱིལ་ཀྲུང་གིས་བཞུགས་པ། དེའི་པང་དུ་ཡུམ་རྡོ་རྗེ་དབྱིངས་ཀྱི་དབང་ཕྱུག་མ་སྔོན་མོ་རྡོ་རྗེ་དང་དྲིལ་བུ་འཛིན་པས་ཡབ་ལ་འཁྱུད་ཅིང་པདྨའི་འདུག་སྟངས་ཀྱིས་སྦྱོར་བ། གཉིས་ཀའང་ཞི་ཞིང་སྒེག་པའི་ཉམས་ཅན་འཛུམ་བག་དང་ལྡན་པ། མཚན་དང་དཔེ་བྱད་ཀྱིས་སྤྲས་ཤིང་རུས་པ་དང་རིན་པོ་ཆེའི་རྒྱན་གྱིས་བརྒྱན་པ། དར་གྱི་ན་བཟས་མཛེས་ཤིང་འོད་ཀྱི་གུར་ཁང་ན་བཞུགས་པ། ཞིང་ཁམས་ཐམས་ཅད་སྐུ་ལ་རྫོགས་ཤིང་སྣང་ལ་རང་བཞིན་མེད་པ། བླ་མ་ཡི་དམ་དཔའ་བོ་མཁའ་འགྲོ་ཆོས་སྐྱོང་སྲུང་མའི་ཚོགས་དཔག་ཏུ་མེད་པས་བསྐོར་བ། དེའི་དཔྲལ་བར་ཟླ་བ་ལ་ཨོཾ་དཀར་པོ། མགྲིན་པར་པདྨ་ལ་ཨཱཿདམར་པོ། ཐུགས་ཀར་ཉི་མ་ལ་ཧཱུྃ་སྔོན་པོས་མཚན་པའི་ཧཱུྃ་ལས་འོད་ཟེར་འཕྲོས་པས། ཕྱོགས་བཅུའི་བླ་མ་ཡི་དམ་དཔའ་བོ་མཁའ་འགྲོ་ཆོས་སྐྱོང་སྲུང་མའི་ཚོགས་དང་བཅས་པ་སྤྱན་དྲངས། རྩ་བའི་བླ་མ་ལ་ཐིམ་པས་བླ་མ་སངས་རྒྱས་ཐམས་ཅད་འདུས་པའི་བདག་ཉིད་དུ་གྱུར། །ཞེས་མོས་ནས། སྐུ་གསུམ་ཡོངས་རྫོགས་བླ་མའི་ཚོགས་རྣམས་ལ། །ཕྱི་ནང་གསང་གསུམ་དེ་བཞིན་ཉིད་ཀྱིས་མཆོད། །བདག་ལུས་ལོངས་སྤྱོད་སྣང་སྲིད་ཡོངས་བཞེས་ནས། །བླ་མེད་མཆོག་གི་དངོས་གྲུབ་སྩལ་དུ་གསོལ། །

ཞེས་སོགས་དངོས་སུ་འབྱོར་པ་དང་ཡིད་ཀྱིས་སྤྲུལ་པའི་མཆོད་པ་མཐའ་ཡས་པ་ཕུལ་ཏེ། སྙིང་ཁུང་རུས་པའི་གཏིང་ནས་མཆི་མ་འཁྲུགས་ཤིང་གཡོ་བ་ཙམ་དུ་འཁོར་བ་ལ་སྐྱོ་ཤས། ཐར་པ་ལ་ངེས་འབྱུང༌། བླ་མ་ལ་མོས་གུས་ཀྱི་ཤུགས་ཚད་མེད་པ་བསྐྱེད་ལ་གདུང་བའི་དབྱངས་ཀྱིས། མ་ནམ་མཁའ་དང་མཉམ་པའི་སེམས་ཅན་ཐམས་ཅད། བླ་མ་འོད་གསལ་ཆོས་སྐུ་ལ་གསོལ་བ་འདེབས་སོ། །བླ་མ་བདེ་ཆེན་ལོངས་སྐུ་ལ་༴ །བླ་མ་ཐུགས་རྗེ་སྤྲུལ་སྐུ་ལ་༴ །བླ་མ་སྐུ་གསུམ་དབྱེར་མེད་ལ་༴ །བླ་མ་སྐུ་བཞིའི་བདག་ཉིད་ལ་༴ །བླ་མ་སྐུ་ལྔ་ཡོངས་རྫོགས་ལ་གསོལ་བ་འདེབས་སོ། །ཞེས་དང༌། དཔལ་ལྡན་བླ་མའི་སྐུ་ལ་གསོལ་བ་འདེབས། །སྒྱུ་ལུས་སྤྲུལ་སྐུར་རྟོགས་པར་བྱིན་གྱིས་རློབས། །དཔལ་ལྡན་བླ་མའི་གསུང་ལ་༴ །སྲོག་རྩོལ་ལོངས་སྐུར་རྟོགས་པར་༴ །དཔལ་ལྡན་བླ་མའི་ཐུགས་ལ་༴ །སེམས་ཉིད་ཆོས་སྐུར་རྟོགས་པར་༴ །ཆོས་ཆོས་སུ་འགྲོ་བར་༴ །ཆོས་ལམ་དུ་ལོངས་པར་༴ །ལམ་གྱི་འཁྲུལ་པ་སེལ་བར་༴ །འཁྲུལ་པ་ཡེ་ཤེས་སུ་འཆར་བར་༴ །གཏུམ་མོ་སྒྱུ་ལུས་གྲུབ་པར་༴ །རྨི་ལམ་འོད་གསལ་ཟིན་པར་༴ །བར་དོ་འཕོ་བ་འབྱོངས་པར་༴ །ཚེ་གཅིག་སངས་རྒྱས་ཐོབ་པར་བྱིན་༴ །ཞེས་རྩེ་གཅིག་ཏུ་གསོལ་བ་གདབ། སྤྲོ་ན་བླ་མ་བརྒྱུད་པའི་གསོལ་འདེབས་ཀྱང་བྱའོ། །

ཞེས་གཅོད་ནང་མ་གཅིག་ལ་གསོལ་བ་བཏབ་སྟེ་བྱིན་རླབས་ཞུས་པ་དེ་འདི་ཡིན། འཁོར་བ་ལ་སྐྱོ་ཤས་དེ་དང་པོ་རེད། འོ་ན་འཁོར་བ་ལ་ཞེན་པ་ལོག་ན། དེའི་ལྡོག་ཕྱོགས་གང་དགོས་སྙམ་ན། ཐར་པ་ལ་བློ་བསྐྱེད་དགོས་པ་ཡིན། ཐར་པ་ལ་ངེས་འབྱུང་ཞེས་ངེས་པར་འབྱུང་བ་ཞེས་གང་ནས་ངེས་པར་འབྱུང་ན། འཁོར་བ་ལས་ངེས་པར་འབྱུང་དགོས་པ་ཡིན། གང་གིས་ངེས་པར་འབྱུང་ན། བདེན་བཞིའི་ཉམས་ལེན་གྱི་ཐོག་ནས་ངེས་པར་འབྱུང་དགོས་པ་ཡིན། གང་དུ་ངེས་པར་འབྱུང་ན། ཐར་པ་དང་ཐམས་ཅད་མཁྱེན་པ་ལ་ངེས་པར་འབྱུང་དགོས་ཞེས་གང་ནས། གང་གིས། གང་དུ་སྟེ་གསུམ་གྱི་ཐོག་ནས་ངེས་པར་འབྱུང་དགོས་པའི་ཚུལ་རེ་མདའ་བས་གསུངས་པ་ལྟ་བུ་རེད། དེ་ལྟར་ངེས་པར་འབྱུང་བ་འཁོར་བ་ལ་སྐྱོ་ཤས་དང་ཐར་པ་ལ་འདོད་པ་སྟེ་གཉིས་གཞི་ལ་བཞག །དེའི་སྟེང་ལམ་ཟབ་མོ་བླ་མའི་རྣལ་འབྱོར་གྱི་ཉམས་ལེན་སྐབས་བླ་མ་ལ་མོས་གུས་དགོས་པ་ཡིན། བླ་མ་ལ་མོས་གུས་ཞེས་པའང་བླ་མ་མཐོང་ཚད་ལ་ང་ཚོས་ཕྱག་འཚལ་བ་དང་ཕྱག་དབང་ཞུ་རྒྱུ་ཞིག་ཡོད་པ་དེ་ལུགས་སྲོལ་ཙམ་ཞིག་རེད། ལྷག་པར་དུ་བོད་པའི་སྤྱི་ཚོགས་ནང་བླ་མ་ཞིག་གི་སྐུ་མདུན་ལ་སླེབས་པ་ན་ཕྱག་ཅིག་འཚལ་མ་བཞག་ན། མི་ཡིས་ཁ་རྡུང་རྒྱག་མི་ཡོང་ངམ། བླ་མ་དེ་དགོངས་པ་འཚོམས་མི་ཡོང་ངམ། ཞབས་ཕྱིས་གཅིག་ལབ་མི་ཡོང་ངམ་བསམ་པའི་བག་ཆགས་ཤིག་ཡོད། བོད་པའི་ཆོས་བྱེད་པ་ཐམས་ཅད་ལ་དཀའ་ངལ་འདི་ཡོད། ཕྱི་རྒྱལ་ལ་ནི་དེ་འདྲ་མི་འདུག །ཕར་ནས་མི་ཞིག་གིས་ཕྱག་འཚལ་ན་དེའི་ལད་ཟློས་བྱས་ཏེ་ད་ལུགས་སྲོལ་ཡིན་ས་རེད་སྙམ་ནས་ཕྱག་འཚལ་རྒྱུ་ཞིག་ཡོད། དེ་ལ་ཡང་ཕན་པ་མེད། ཡང་འབྲེལ་ལམ་གྱི་ཐོག་ནས་རྟག་པར་མཉམ་དུ་གཤིབ་དགོས་དུས་ཕྱག་ཅིག་མ་འཚལ་ན་ཁོའི་སེམས་ལ་གནོད་མི་ཡོང་ངམ་བསམ་ནས་ཕྱག་འཚལ་རྒྱུ་ཞིག་དང༌། ཡང་བོད་ཀྱི་བླ་མ་ལ་ཕྱག་མ་འཚལ་ན་སྐྱོན་ཡོང་གི་རེད་ཅེས་དྭོགས་ནས་འཇིགས་སྐྲག་གི་ཐོག་ནས་ཕྱག་འཚལ་རྒྱུ་ཞིག་ཡོད། འོ་དེ་ཡང་ཕན་ཡོན་མེད། ཡང་བླ་མ་ལ་བསྟོད་ར་བཏང་བའི་སྐབས་སྐྱབས་རྗེ་དང་འདི་ཕྱི་གཉིས་ཀྱི་སྐྱབས་གནས་ཞེས་པ་ལྟ་བུ་དང༌། སངས་རྒྱས་ཐམས་ཅད་འདུས་པའི་ངོ་བོ་ཞེས་བརྗོད་ཀྱི་ཡོད་རེད་དེ། ཡིན་ན་ཡང་ཤེས་ནས་དད་པ་ཐོབ་པ་མེད་པ་ཡིན་དུས། ལུགས་སྲོལ་གྱི་ཐོག་ནས་ཕྱག་འཚལ་ཡོང་དུས་དོན་མེད་སོ་སོའི་གཟུགས་པོ་ལ་རྔུལ་ནག་རྒྱག་པའི་དཀའ་ངལ་མ་གཏོགས་དེ་ལ་དགོས་པ་དང་ཕན་ཡོན་མེད། མོས་གུས་ཞེས་པ་དེ་རྒྱུ་མཚན་ཤེས་པའི་ཐོག་ནས་བླ་མའི་ཡོན་ཏན་ལ་བསམ་མནོ་བཏང༌། སྐུ་ཡི་ཡོན་ཏན་གསང་བ་བསམ་གྱིས་མི་ཁྱབ་པ། གསུང་གི་ཡོན་ཏན་གསང་བ་བསམ་གྱིས་མི་ཁྱབ་པ། ཐུགས་ཀྱི་ཡོན་ཏན་གསང་བ་བསམ་གྱིས་མི་ཁྱབ་པ། ཡོན་ཏན་དང་ཕྲིན་ལས་ཀྱི་གསང་བ་བསམ་གྱིས་མི་ཁྱབ་པའི་ཚུལ་དེ་དག་ཤེས་ཡོང་དུས། དང་པོ་དེ་བླ་མའི་མདུན་དུ་གདམས་ངག་དང་མན་ངག་ཞུ། འོད་གསལ་ཕྱག་རྒྱ་ཆེན་པོའི་གནས་ལུགས་སྟོན་པའི་གདམས་ངག་དེ་དག་ཞུ། དེ་སོ་སོས་སེམས་ལ་ཉམས་ལེན་བྱེད་ཡོང་དུས་ཉམས་ལེན་གྱི་སྟོབས་ཀྱིས་རིམ་གྱིས་ཉོན་མོངས་པ་ཐམས་ཅད་མར་བྲི་རུ་བྲི་རུ་སོང༌། ཡོན་ཏན་གྱི་ཕྱོགས་ཐམས་ཅད་ཡར་གོང་ནས་གོང་དུ་འཕེལ་ཡོང་དུས། བླ་མ་དེའི་མདུན་དུ་གདམས་ངག་འདི་ཞུས་ཏེ་སེམས་རྒྱུད་ལ་སྦྱར་བས་རང་རྒྱུད་ཀྱི་ཉོན་མོངས་དེ་དག་མར་ཞི། ཡོན་ཏན་གྱི་ཚོགས་ཐམས་ཅད་གོང་དུ་འཕེལ་ཐུབ་པའི་ནུས་པ་དེ་བླ་མ་དེ་ལ་བསྟེན་ནས་བྱུང༌། བླ་མ་འདི་གཞན་དང་མི་འདྲ་བ་ཞིག་ཡིན་སྙམ་པའི་མོས་གུས་སྐྱེ་དགོས་པ་ཡིན། གསུང་ཚད་མ་དང་གདམས་ངག་ཚད་མ་ལ་བརྟེན་ནས་བླ་མ་ལ་ཡིད་ཆེས་སྐྱེ་དགོས་པ་མ་གཏོགས། ཕ་ཕོར་རུ་སྒོ་སྨྲ་བརྗོད། རིགས་རུས་དང་ཆོས་ལུགས། ལུང་ཚན་སོགས་ཀྱི་ཐོག་ལ་བསམ་སྟེ་བླ་མ་ལ་སྐྱབས་སུ་འགྲོ་བ་དང་མོས་གུས་སོགས་སྐྱེ་དགོས་ན། དཀའ་ལས་ཁག་པོ་རེད། གལ་ཏེ་འདི་འདྲ་ཞིག་ལ་ངོས་འཛིན་བྱས་ན་ཡང་ངོས་འཛིན་བྱེད་ས་ནོར་བ་ཡིན། དེའི་ཕྱིར་ན་མོས་གུས་བཅོས་མ་མ་ཡིན་པ་ཞེས་བཅོས་མ་མ་ཡིན་པའི་མོས་གུས་སྐྱེ་བའི་གཞི་དེ་གང་ལ་བསྟེན་ནས་ཡོང་དགོས་ན། ཆོས་དང་གདམས་ངག་ལ་བསྟེན་དགོས་པ། མན་ངག་དང་ཉམས་ལེན་ལ་བསྟེན་དགོས་པ་ཡིན། དེ་ལྟའི་སྟེང་ནས་སོ་སོས་སློབ་སྦྱོང་དང་ཉམས་ལེན་བྱས་པས། དེའིསྟོབས་ཀྱིས་དྲངས་པའི་མོས་གུས་དེ་ལ་འདི་ནས་བྱང་ཆུབ་མ་ཐོབ་ཀྱི་བར་དུ་འཕོ་འགྱུར་མེད་པ། བྱང་ཆུབ་མ་ཐོབ་ཀྱི་བར་མོས་གུས་ལ་ཐང་ལྷོད་མེད་པ། འདི་ནས་བྱང་ཆུབ་མ་ཐོབ་ཀྱི་བར་མོས་གུས་དང་བྲལ་བ་མེད་པ་སྟེ་བཅོས་མ་མ་ཡིན་པའི་མོས་གུས་བྲག་ལ་རི་མོ་བྲིས་པ་ལྟ་བུར་གནས་ཐུབ་པའི་རྒྱུ་མཚན་ཡང་ཉམས་ལེན་དང་གདམས་ངག་གི་སྟེང་ནས་ཡོང་བ་ཡིན་པས་གདམས་ངག་ཞུ་དགོས་གསུངས་པ་འདི་ཡིན། དེའི་སྟེང་ནས་བླ་མ་ལ་མོས་གུས་ཤུགས་ཀྱིས་འཕེལ་བར་བརྟེན་འདིར། སྒྱུ་ལུས་སྤྲུལ་སྐུར་རྟོགས་པར་བྱིན་གྱིས་རློབས། །སྲོག་རྩོལ་ལོངས་སྐུར་རྟོགས་པར་བྱིན་གྱིས་རློབས། །སེམས་ཉིད་ཆོས་སྐུར་རྟོགས་པར་བྱིན་གྱིས་རློབས། །ཆོས་ཆོས་སུ་འགྲོ། ཆོས་ལམ་དུ་ལོངས། ལམ་གྱིས་འཁྲུལ་པ་སེལ་བ། འཁྲུལ་པ་ཡེ་ཤེས་སུ་འཆར་བ། གཏུམ་མོ་སྒྱུ་ལུས་གྲུབ་པ། རྨི་ལམ་འོད་གསལ་དུ་ཟིན་ཐུབ་པ། བར་དོ་ལ་འཕོ་བ་འབྱོངས་ཐུབ་པ་དང༌། དེ་ལ་བརྟེན་ནས་ཚེ་གཅིག་སངས་རྒྱས་ཐོབ་པ་དེ་དག་ཐམས་ཅད་བླ་མ་ལ་བསྟེན་ནས་ཡོང་གི་ཡོད། །
ཏིང་འཛིན་གྱི་དབང་བླང་བ་ནི། དེ་ལྟར་གསོལ་བ་བཏབ་པས་དཔལ་ལྡན་བླ་མའི་དཔྲལ་བའི་ཨོཾ་དཀར་པོ་ལས་འོད་ཟེར་དཀར་པོ་འཕྲོས། རང་གི་དཔྲལ་བར་ཐིམ། བུམ་པའི་དབང་ཐོབ། ལུས་ཀྱི་སྒྲིབ་པ་སྦྱངས། ལམ་བསྐྱེད་རིམ་སྒོམ་པ་ལ་དབང༌། འབྲས་བུ་སྤྲུལ་སྐུ་འགྲུབ་པར་བྱས། བླ་མའི་མགྲིན་པའི་ཨཱཿདམར་པོ་ལས་འོད་ཟེར་དམར་པོ་འཕྲོས། རང་གི་མགྲིན་པར་ཐིམ། གསང་བའི་དབང་ཐོབ། ངག་གི་སྒྲིབ་པ་སྦྱངས། ལམ་རང་བྱིན་གྱིས་བརླབས་ཀྱི་རྫོགས་རིམ་སྒོམ་པ་ལ་དབང༌། འབྲས་བུ་ལོངས་སྐུ་འགྲུབ་པར་བྱས། བླ་མའི་ཐུགས་ཀའི་ཧཱུྃ་སྔོན་པོ་ལས་འོད་ཟེར་སྔོན་པོ་འཕྲོས། རང་གི་སྙིང་གར་ཐིམ། ཤེས་རབ་ཡེ་ཤེས་ཀྱི་དབང་ཐོབ། ཡིད་ཀྱི་སྒྲིབ་པ་སྦྱངས། ལམ་ཕྱག་རྒྱ་ཆེན་པོ་སྒོམ་པ་ལ་དབང༌། འབྲས་བུ་ཆོས་སྐུ་འགྲུབ་པར་བྱས། བླ་མའི་གནས་གསུམ་གྱི་ཡི་གེ་གསུམ་ལས་འོད་ཟེར་དཀར་དམར་མཐིང་གསུམ་འཕྲོས། རང་གི་གནས་གསུམ་དུ་ཐིམ། དབང་བཞི་པ་ཐོབ། ལུས་ངག་ཡིད་གསུམ་གྱི་སྒྲིབ་པའི་བག་ཆགས་སྦྱངས། ལམ་བདེན་གཉིས་དབྱེར་མེད་ཟུང་འཇུག་གི་རྫོགས་རིམ་སྒོམ་པ་ལ་དབང༌། འབྲས་བུ་ངོ་བོ་ཉིད་ཀྱི་སྐུ་འགྲུབ་པར་བྱས། ཞེས་མོས།
ཞེས་པ་སྟེ་དབང་བཞི་ལེན་པའོ། །སྐུ་གསུང་ཐུགས། དེ་ནས་གཅིག་ཆ་མཉམ་གྱི་ཡོན་ཏན། ཉམས་ལེན་བྱས་ཏེ་བླ་མ་སྒྲུབ་ནས་གྲུབ་དགོས་པ་ཡིན། འོ་ན་བླ་མ་གྲུབ་པ་ལ་གང་སྒྲུབ་དགོས་སྙམ་ན། བླ་མ་ལ་བཅོས་མ་མ་ཡིན་པའི་དད་པ་དང་མོས་གུས་ཀྱིས་བླ་མ་སྒྲུབ་དགོས། དེའི་ཉམས་ལེན་ལ་གནད་དུ་བསྣུན་ནས་བླ་མ་སྒྲུབ་པའི་གྲུབ་དོན་ལ་གང་འཐོབ་དགོས་ན། བླ་མ་རང་སེམས་སུ་མཐོང་དགོས་པ། བླ་མ་སངས་རྒྱས་སེམས་ལ་རྙེད་དགོས་པ་ཡིན། བླ་མ་ཆོས་སྐུར་མཐོང༌། བླ་མ་སེམས་ཉིད་ཀྱི་ངོ་བོ་ལ་མཐོང༌། དེར་རྟོགས་པ་དང༌། དེ་རུ་སྒྲོ་འདོགས་ཆོད་ཐུབ་ན། བླ་མ་སྒྲུབ་པའི་གྲུབ་དོན་འདི་ཡིན། རྒྱལ་བ་རྒོད་ཚང་པས་ཀྱང༌།

ཞེས་མི་བདག་གིས་བླ་མ་བསྒྲུབས། བླ་མའི་རྣལ་འབྱོར་ཉམས་ལེན་བྱས། བླ་མ་ལ་བསྟེན། གདམས་ངག་ཞུས། ཉམས་ལེན་བྱས་ཏེ་བླ་མ་བསྒྲུབས་པ་ཡིན་གསུངས། སྒྲུབ་ནས་གྲུབ་སོང་ཞེས་གང་གྲུབ་ཅེ་ན། བླ་མ་ཆོས་སྐུའི་ཞལ་ཡང་མཐོང༌། ཞེས་གནས་ལུགས་དོན་དམ་པའི་བླ་མ་མཐོང་སོང་གསུངས་པ་ཡིན། གནས་ལུགས་དོན་དམ་པའི་བླ་མ་དེ་གང་སྙམ་ན། རང་གི་སེམས་ཡིན། གསལ་སྟོང་དབྱེར་མེད་ཟེར་ནའང་འདྲ། བདེ་སྟོང་དབྱེར་མེད་དང་སྣང་སྟོང་དབྱེར་མེད་སོགས་གང་སྨྲ་ན་ཡང་འདྲ། གང་ལྟར་གནས་ལུགས་འོད་གསལ་ལྷན་ཅིག་སྐྱེས་པའི་ཕྱག་རྒྱ་ཆེན་པོའི་འོད་གསལ་སེམས་ཀྱི་གནས་ལུགས་མཐོང་ན། བླ་མ་མཐོང་བ་རེད། དོན་དམ་གྱི་བླ་མ་ཡང་དེ་རེད་ལ་གནས་ལུགས་ཀྱི་བླ་མ་ཡང་དེ་རེད། ད་ལྟའི་བར་བླ་ལ་ཁྱེད་ལ་སྐྱབས་སུ་མཆིའོ། །བྱིན་གྱིས་བརླབ་ཏུ་གསོལ། ཁྱེད་ལ་བྱང་ཆུབ་མ་ཐོབ་ཀྱི་བར་དུ་ཐུགས་དང་འབྲལ་བ་མེད་པར་ཤོག་ཅེས་སྨོན་ལམ་བརྒྱབ་པའི་འབྲས་བུར་འདི་གྲུབ་དགོས་པ་ཡིན། བླ་མ་སངས་རྒྱས་དངོས་སུ་མཐོང་དགོས་པ། བླ་མ་ལྷན་ཅིག་སྐྱེས་པའི་ཡེ་ཤེས་ཀྱི་ངོ་བོར་མཐོང་དགོས་པ། བླ་མ་ཆོས་སྐུ་རུ་མཐོང་དགོས་པའི་ཕྱིར་ན། མི་བདག་གིས་བླ་མ་སྒྲུབ་པས་གྲུབ། །བླ་མ་ཆོས་སྐུའི་ཞལ་ཡང་མཐོང༌། །ཞེས་པའོ། །སྣང་སྲིད་བླ་མའི་རོལ་པ་ཤར། །ཞེས་སྣང་སྲིད་བླ་མའི་རོལ་པར་ཤར་བ་སྟེ་སྣང་བ་ཐམས་ཅད་བླ་མ། སྒྲ་གྲག་ཚད་ཐམས་ཅད་བླ་མའི་གསུང༌། དྲན་རྟོག་ཐམས་ཅད་བླ་མའི་ཐུགས་ཀྱི་ཡེ་ཤེས་ཀྱི་རོལ་པ་ཞེས་སྔགས་ཀྱི་ཉམས་ལེན་ནང་འདི་རང་ཡོང་གི་ཡོད། སྒྲ་གྲག་ཚད་ཐམས་ཅད་བླ་མའི་གསུང་དང༌། གཡོ་ཚད་ཐམས་ཅད་བླ་མའི་སྐུ། དྲན་རྟོག་ཐམས་ཅད་བླ་མའི་ཐུགས་ཀྱི་ཡེ་ཤེས་ཀྱི་རོལ་པར་ཤར་ཐུབ་ན། ཁྱེད་ནི་བླ་མ་ཁྱེད་ནི་ཡི་དམ་ཁྱེད་ནི་མཁའ་འགྲོ་ཆོས་སྐྱོང་སྟེ། ཞེས་བླ་མ་ཁྱེད་ཀྱི་ནང་ལ་ཡི་དམ་ཚང༌། དཔའ་བོ་ཡང་ཚང༌། དཔའ་མོ་ཡང་ཚང༌། ཆོས་སྐྱོང་ཡང་ཚང་ཡོད་ཅེས་གསུངས་པ་ལྟ་བུ་ཡིན། སྣང་བ་ཐམས་ཅད་བླ་མའི་སྐུ་ལ་ཤར་ན། ཡི་དམ་དང་ཆོས་སྐྱོང་དཔའ་བོ་མཁའ་འགྲོ་ཡང་དེའི་ནང་ཚང་བ། སྒྲ་གྲག་ཚད་ཐམས་ཅད་བླ་མའི་གསུང་ལ་ཤེས་ཐུབ་ན། ཡི་དམ་དཔའ་བོ་མཁའ་འགྲོ་ཆོས་སྐྱོང་སྲུང་མ་ཡང་དེར་ཚང་བ། དྲན་རྟོག་ཐམས་ཅད་བླ་མའི་ཐུགས་དང་ཡེ་ཤེས་ཀྱི་རོལ་པར་ཤར་ཐུབ་ན། དཔའ་བོ་མཁའ་འགྲོ་ཡི་དམ་ཆོས་སྐྱོང་སྲུང་མ་ཡང་དེར་ཚང་བ་ཡིན། འདི་ནི་བླ་མའོ། །འདི་ནི་ཡི་དམ་མོ། །འདི་ནི་ཆོས་སྐྱོང་ངོ་ཞེས་སོ་སོར་མེད་པ་ཡིན། སྣང་སྲིད་བླ་མའི་རོལ་པར་ཤར་བའི་རྒྱུ་མཚན་དེ་གོང་དུ་རྒྱལ་བ་རྒོད་ཚང་པས་གསུངས་པ་ལྟ་བུ་རེད། ཡུལ་དངོས་པོར་འཛིན་པའི་ཞེན་པ་བྲལ། །ཞེས་པ་སྟེ་ཡུལ་མཛེས་པ་ལ་ཆགས། མི་མཛེས་པ་ལ་སྡང༌། དེ་བཞིན་དུ་སྙན་པ་ལ་ཆགས། མ་སྙན་པ་ལ་སྡང་བ་སོགས་ཆགས་སྡང་གི་བློ་རང་སར་ཞི་བ་ཞེས་པ་སྟེ་འདི་ཡིན། དེ་ཡང་ཆེད་དུ་གཉེན་པོར་བསྟེན་ཏེ་སྤང་བྱ་བཅོམ་མི་དགོས་པར་གནས་ལུགས་ལྷན་ཅིག་སྐྱེས་པའི་འོད་གསལ་ཕྱག་རྒྱ་ཆེན་པོའི་ངོ་བོ་མཐོང་བས་ཡུལ་ལ་འཛིན་པ་དང་ཞེན་པ་སོགས་ཆགས་སྡང་གི་བློ་མཐའ་དག་རང་སར་ཞི་བ། རླུང་ཞི་བས་རྦ་རླབས་མི་ཞི། རྦ་རླབས་རང་སར་ཞི་བས་རྒྱ་མཚོའམ་ཆུ་བོ་ངང་གིས་གནས་པ་བཞིན་སྣང་སྲིད་ཀྱི་སྤྲོ་བསྡུ་ཐམས་ཅད་ཆོས་ཀྱི་དབྱིངས་ཀྱི་ནང་འབྱམས་པ་ན། དེ་ཚེ་གནས་ལུགས་ངོ་མ་དོན་དམ་པའི་བླ་མ་མཇལ་བ་ཡིན། དོན་དམ་གྱི་བླ་མའི་རྣལ་འབྱོར་ཀྱང་འདི་ཡིན་ནོ། །

ཞེས་པའི་གསོལ་འདེབས་ཚིག་རྐང་དྲུག་མ་འདི་འདོན། འདོན་ནས་བླ་མ་རང་སེམས་སུ་དབྱེར་མ་ཕྱེད་པར་ཤེས་པའི་ངང་ནས་མཉམ་པར་གཞག་སྟེ་སེམས་ངོ་བོ་སྟོང་ལ་རང་བཞིན་གསལ་བ། གསལ་ཧྲིག་གེ་བ། གསལ་སིང་ངེ་བའི་ངང་ལ་སེམས་སྦུག་སྲུབ་དོག་པོར་གྱུར་བའི་རྐྱེན་དེ་ག་འདྲ་ཞིག་འདུག །མ་བདེ་བར་ཆགས་རྒྱུ་དེ་གང་འདྲ་ཞིག་འདུག །རྣམ་རྟོག་ངན་པ་དེ་ཅི་འདྲ་ཞིག་འདུག་ཅེས་བརྟག་ཅིང་དཔྱད། དཔྱད་ཡོང་དུས་ཙ་ན་རྙེད་རྒྱུ་མེད་པ་ངོས་བཟུང་དང་བྲལ་བ་རེད། ངོས་བཟུང་མེད་པའི་གནས་ལུགས་སེམས་ཀྱི་ངང་ལ་སྐར་མ་ལྔ་དྲུག་ཡུན་ལ་མཉམ་པར་བཞག །སྤྱིར་སྒོམ་ཞེས་པ་དེ་ང་ཚོས་ཚད་མཐོན་པོ་ཞིག་ལ་ངོས་འཛིན། སྒོམ་རྒྱག་དགོས་ཟེར་ཙ་ན་ཚད་མཐོན་པོ་ཞིག་གི་མནོ་བསམ་འཁོར་གྱིན་ཡོད། དེ་འདྲ་གང་ཡང་མ་རེད། སེམས་ལྷོད་ལྷོད་བྱེད་དེ་མ་བཅོས་པའི་ངང་ལ་མཉམ་པར་བཞག་ན་དེ་སྒོམ་རེད། སེམས་ཀྱི་ངོ་བོ་གསལ་སྟོང་དབྱེར་མེད་ཡིན་པ་དེ་ང་ཚོས་རྟག་པར་མཐོང་མཐོང་བ་རེད། ངོས་བཟུང་མ་བཟུང་གི་ཁྱད་པར་ཙམ་མ་གཏོགས་དུས་རྟག་ཏུ་མཐོང་མཐོང་བ་རེད། གནས་ལུགས་དེའི་ངང་ལ་བཅས་བཅོས་མེད་པར་བླ་མའི་མན་ངག་དང་འགྲོགས་ཏེ་དར་ཞིག་མཉམ་པར་བཞག་གི་ཡིན་ན། དེ་ནས་སེམས་སྦུག་སྲུབ་དོག་པོ་ཡོད་པ་དེ་མར་ཞི། རྣམ་རྟོག་འཁྲུག་ཆ་ཆེན་པོར་ཡོད་པ་དེ་ཡང་མར་ཞི་ནས་དེའི་ངང་ལ་ཞི་བདེ་དང་བག་ཕེབས་པོ་ཞིག་ཡོང་གི་ཡོད། དེའི་ངང་ནས་ག་ལེ་ག་ལེར་གནས་ལུགས་ཀྱི་ནང་འགྲོ་བའི་གོ་སྐབས་ཀྱང་ཡོང་གི་རེད། ལམ་ལ་བསླབ་པའི་སྐབས་འདི་ལྟ་བུར་བསླབ་དགོས་པ་ཡིན་ནོ། །
དེ་ནས་བསྐྱེད་རིམ་ལ་གཙོ་བོར་གཞོལ་བས་ཡི་དམ་གྱི་ལྷ་བསྒོམ་པ་དང་སྔགས་བཟླ་བ་སོགས་བྱ། སྤྲོས་བཅས་ཀྱི་རྫོགས་རིམ་ལ་གཞོལ་བས་གཏུམ་མོ་ལ་སོགས་པའི་དམིགས་རིམ་བསྒོམ། སྤྲོས་མེད་ལ་གཞོལ་བས་ཕྱག་རྒྱ་ཆེན་པོ་ལ་མཉམ་པར་བཞག་ལ། ཐུན་ལས་ལྡང་བའི་ཚེ། བསོད་ནམས་འདི་ཡིས་ཐམས་ཅད་གཟིགས་པ་ཉིད། །ཐོབ་ནས་ཉེས་པའི་དགྲ་རྣམས་ཕམ་བྱས་ཏེ། །སྐྱེ་རྒ་ན་འཆིའི་རྦ་ཀློང་འཁྲུགས་པ་ཡི། །སྲིད་པའི་མཚོ་ལས་འགྲོ་བ་སྒྲོལ་བར་ཤོག །ཅེས་སོགས་བསྔོ་བས་རྒྱས་བཏབ་སྟེ་ཁ་ན་མ་ཐོ་བ་མེད་པར་གཏོར་མའི་རིམ་པ་ལ་སོགས་ཐུན་མཚམས་ཀྱི་རྣལ་འབྱོར་ལ་འཇུག་པར་བྱའོ། །
ཞེས་ཐོ་རངས་ཡར་ལངས་དུས་ཉིད་ནས་དང་ཤོས་སྔོན་འགྲོའི་ཆོས་བཞིའི་ཉམས་ལེན་དེ་གལ་ཆེ་ཤོས་ཡིན་པར་གསུང་གི་འདུག སྔོན་འགྲོའི་ཆོས་བཞི་ཉམས་ལེན་ཞིག་བྱས། དེ་ནས་ཁ་ན་མ་ཐོ་བ་མེད་པའི་དགེ་སྦྱོར་ཐམས་ཅད་བྱས། ཁ་ན་མ་ཐོ་བ་ཞེ་ན། ཕྱིན་ཅི་ལོག་པ། ཡང་དག་པ་མ་ཡིན་པ། སྐྱོན་དང་བཅས་པ། ནོར་འཁྲུལ་དང་བཅས་པ། རྣམ་པར་དག་པ་མ་ཡིན་པ་དེ་དག་ཡིན་ལ། རྣམ་པར་དག་པ་མ་ཡིན་པའི་དགེ་བའམ་ཆོས་གཙོ་བོ་དེ་གང་ཡིན་ཞེ་ན། བདག་ཏུ་འཛིན་པ་དང་བདེན་པར་འཛིན་པ། ཚེ་འདི་དང་ཕྱིན་ཅི་ལོག་པའི་ཉོན་མོངས་པའི་རྟོག་པ་དང་འབྲེལ་ན། རྣམ་པར་དག་པ་མ་ཡིན་པའི་དགེ་བ་ཡིན། དེ་ལྟ་བུ་མ་ཡིན་པར་སྔོན་འགྲོའི་ཉམས་ལེན་གྱི་བློ་སྔོན་དུ་སོང་ན། བློ་དེས་དྲངས་པའི་ཆོས་སྤྱོད་ཀྱི་རིམ་པ་ཀུན་དགེ་བ་ལ་འགྲོ་བ་ཡིན། དེའི་ཕྱིར་ན་ཐུན་མཚམས་ཀྱི་རྣལ་འབྱོར་ལ་འཇུག་པར་བྱའོ་ཞེས་གསུངས་པའོ། །
